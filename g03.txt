/* ===== INLINED: cpu-coach-ai.js ===== */
/* ============================================================================
 * CPU COACH AI — Courtside 27
 * ----------------------------------------------------------------------------
 * A strategic opponent coach that plays "figurative chess" against the user.
 * It watches what you run, recognizes patterns, and counters — then tells
 * you what it's doing so the chess match is visible.
 *
 * DESIGN PHILOSOPHY
 *  - Every decision traces to observed data, never a raw coin flip.
 *  - Personality (grinder vs. blur) biases thresholds and scheme preferences.
 *  - Human-like imperfection: occasional stubbornness, delayed reactions.
 *  - All announcements go through api.announce() so the user SEES the counter.
 *
 * INTEGRATION
 *  const coach = new CPUCoach(gameAPI, { personality: 'grinder' });
 *  coach.onGameStart();
 *  // each frame: coach.update(dt);
 *  // on events: coach.onPossessionStart(team); coach.onShotResolved(info);
 *  // at half:   coach.onHalftime();
 *
 * The gameAPI adapter is documented at the bottom of this file.
 * ========================================================================== */

class CPUCoach {
  constructor(api, options = {}) {
    this.api = api;

    // ---- Personality -------------------------------------------------------
    // 'grinder' (Pat Riley): half-court, protect paint, slow pace, patient.
    // 'blur'    (Mike D'Antoni): pressure, fast pace, trap, live with chaos.
    // If not specified, pick from the CPU team's identity (passed via options
    // or derived from api). Falls back to a coin flip with a paper trail.
    this.personality = options.personality || this._draftPersonality();
    this.P = CPUCoach.PERSONALITIES[this.personality];

    // ---- Internal state ----------------------------------------------------
    this.timeoutsLeft = 7;
    this.scheme = 'man';              // current defensive scheme
    this.schemeCooldown = 0;          // seconds before scheme may change again
    this.timeoutCooldown = 0;
    this.subCooldown = 0;
    this.lastRunCheck = { user: 0, cpu: 0 };
    this.halftimeDone = false;
    this.gameStarted = false;

    // ---- Pattern memory (sliding windows) ----------------------------------
    this.playHistory = [];            // {play:'pick-roll'|'isolation'|'motion', t}
    this.shotHistory = [];            // {team, points:2|3, kind, made, handler, dist, t}
    this.userRun = 0;                 // current unanswered user scoring run
    this.cpuRun = 0;
    this.lastUserScore = 0;
    this.lastCpuScore = 0;

    // ---- Per-half aggregates -----------------------------------------------
    this.half = this._freshHalf();

    // ---- Rotation ----------------------------------------------------------
    this.cpuMinutes = [0, 0, 0, 0, 0]; // crude minutes tracker per starter slot
    this.hotHand = { idx: -1, points: 0, windowStart: 0 };

    // ---- Late game ----------------------------------------------------------
    this.foulStrategyArmed = false;
    this.lastShotHeld = false;

    // ---- Narrative ----------------------------------------------------------
    this.announced = new Set();        // dedupe one-time announcements
  }

  /* ========================================================================
   * PERSONALITY
   * ====================================================================== */
  static PERSONALITIES = {
    grinder: {
      label: 'GRINDER',
      coachName: 'RILEY',
      // thresholds
      runTimeoutAt: 8,          // call TO when user run hits this
      hedgeAfterPnR: 3,         // hedge after N pick-rolls in window
      trapStarAfter: 12,        // trap after star scores N in a quarter-ish
      zoneAfterThrees: 4,       // zone after N made threes in window
      doublePostAfter: 3,       // double the post after N post touches
      fatigueSubAt: 32,         // sub when stamina below this
      foulTroubleAt: 4,         // protective sub at this many fouls (early)
      // biases
      prefersZone: true,
      prefersTrap: false,
      pace: 'slow',
      pressLikelihood: 0.15,
      stubbornness: 0.22,       // chance to ignore a counter-signal once
      adjustmentDelay: 2,       // extra possessions before reacting
    },
    blur: {
      label: 'BLUR',
      coachName: 'DANTONI',
      runTimeoutAt: 10,
      hedgeAfterPnR: 2,
      trapStarAfter: 10,
      zoneAfterThrees: 5,
      doublePostAfter: 4,
      fatigueSubAt: 26,
      foulTroubleAt: 5,
      prefersZone: false,
      prefersTrap: true,
      pace: 'fast',
      pressLikelihood: 0.65,
      stubbornness: 0.10,
      adjustmentDelay: 0,
    },
  };

  _draftPersonality() {
    // Derive from CPU team identity if the api exposes it; otherwise flip.
    try {
      const id = (this.api.getCpuTeamId && this.api.getCpuTeamId() || '').toLowerCase();
      if (/heat|celtics|spurs|bulls/.test(id)) return 'grinder';
      if (/warriors|suns|nuggets/.test(id)) return 'blur';
    } catch (e) { /* fall through */ }
    return Math.random() < 0.5 ? 'grinder' : 'blur';
  }

  _freshHalf() {
    return {
      userThreesAttempted: 0, userThreesMade: 0,
      userPaintPoints: 0, userMidPoints: 0,
      userPnR: 0, userIso: 0, userPostTouches: 0,
      userTurnovers: 0,
      cpuThreesAttempted: 0, cpuThreesMade: 0,
      startTime: 0,
    };
  }

  /* ========================================================================
   * LIFECYCLE
   * ====================================================================== */
  onGameStart() {
    this.gameStarted = true;
    this.timeoutsLeft = 7;
    this.scheme = 'man';
    this.half = this._freshHalf();
    this.api.announce(
      `${this.P.coachName} BALL`,
      `CPU playing ${this.P.label} style — ${this.P.pace==='slow' ? 'half-court grind, paint protection' : 'pace, pressure, controlled chaos'}.`
    );
    this.api.applyDefensiveScheme('man', { silent: true });
    this.api.setCpuTempo(this.P.pace);
  }

  update(dt) {
    if (!this.gameStarted || this.api.isGameOver()) return;
    this.schemeCooldown = Math.max(0, this.schemeCooldown - dt);
    this.timeoutCooldown = Math.max(0, this.timeoutCooldown - dt);
    this.subCooldown = Math.max(0, this.subCooldown - dt);

    this._trackRuns();
    this._recognizePatterns();
    this._updateDefense();
    this._manageTimeouts();
    this._manageRotation(dt);
    this._manageLateGame();
  }

  onPossessionStart(team) {
    // Called by the host game whenever possession changes.
    this._pruneHistories();
    if (team === 'user') {
      const recent = this.playHistory.slice(-1)[0];
      if (recent) this._notePlayUsage(recent.play);
    }
    // Late-game: decide whether to hold for last shot BEFORE the possession.
    this._considerLastShotHold(team);
  }

  onPlayCall(play) {
    // Host game calls this when the USER changes their offensive set.
    this.playHistory.push({ play, t: this.api.getGameSeconds() });
    if (this.playHistory.length > 40) this.playHistory.shift();
  }

  onShotResolved(info) {
    // info: {team:'user'|'cpu', points:2|3, kind:'LAYUP'|'JUMPER', made:bool,
    //        handler:int, dist:number}
    this.shotHistory.push({ ...info, t: this.api.getGameSeconds() });
    if (this.shotHistory.length > 60) this.shotHistory.shift();

    if (info.team === 'user') {
      if (info.points === 3) {
        this.half.userThreesAttempted++;
        if (info.made) this.half.userThreesMade++;
      } else if (info.kind === 'LAYUP' || info.dist < 3.4) {
        if (info.made) this.half.userPaintPoints += info.points;
      } else {
        if (info.made) this.half.userMidPoints += info.points;
      }
      // Hot-hand tracking for the USER's players (so CPU can key on them).
      if (info.made) this._noteUserScorer(info.handler, info.points);
    } else {
      if (info.points === 3) {
        this.half.cpuThreesAttempted++;
        if (info.made) this.half.cpuThreesMade++;
      }
      if (info.made) this._noteCpuScorer(info.handler, info.points);
    }
  }

  onPostTouch(playerIdx) {
    this.half.userPostTouches++;
    this._postTouchLog = this._postTouchLog || [];
    this._postTouchLog.push({ idx: playerIdx, t: this.api.getGameSeconds() });
  }

  onHalftime() {
    if (this.halftimeDone) return;
    this.halftimeDone = true;
    this._applyHalftimeAdjustments();
    this.half = this._freshHalf(); // reset for second half
  }

  /* ========================================================================
   * PATTERN RECOGNITION
   * ====================================================================== */
  _pruneHistories() {
    // Keep roughly the last ~4 minutes of game time in the window.
    const cutoff = this.api.getGameSeconds() - 240;
    this.playHistory = this.playHistory.filter(p => p.t > cutoff);
    this.shotHistory = this.shotHistory.filter(s => s.t > cutoff);
  }

  _countPlays(play, windowN = 12) {
    const w = this.playHistory.slice(-windowN);
    return w.filter(p => p.play === play).length;
  }

  _recognizePatterns() {
    const p = this.patterns = {
      pnrRate: this._countPlays('pick-roll'),
      isoRate: this._countPlays('isolation'),
      threeRate: this.shotHistory.filter(s => s.team === 'user' && s.points === 3).length,
      threeMakes: this.shotHistory.filter(s => s.team === 'user' && s.points === 3 && s.made).length,
      paintTouches: (this._postTouchLog || []).length,
      userStarIdx: this._userStarIdx,
      userStarPts: this._userStarPts || 0,
    };
    return p;
  }

  _notePlayUsage(play) {
    if (play === 'pick-roll') this.half.userPnR++;
    if (play === 'isolation') this.half.userIso++;
  }

  _noteUserScorer(idx, pts) {
    this._userScores = this._userScores || {};
    this._userScores[idx] = (this._userScores[idx] || 0) + pts;
    // Recompute star.
    let best = -1, bestPts = 0;
    for (const [k, v] of Object.entries(this._userScores)) {
      if (v > bestPts) { bestPts = v; best = +k; }
    }
    this._userStarIdx = best;
    this._userStarPts = bestPts;
  }

  _noteCpuScorer(idx, pts) {
    // Hot hand: who is cooking for the CPU right now.
    const now = this.api.getGameSeconds();
    if (now - this.hotHand.windowStart > 180 || this.hotHand.idx === -1) {
      this.hotHand = { idx, points: pts, windowStart: now };
    } else if (this.hotHand.idx === idx) {
      this.hotHand.points += pts;
    } else {
      // Someone else scored; only dethrone after a clear gap.
      this._challenger = this._challenger || {};
      this._challenger[idx] = (this._challenger[idx] || 0) + pts;
      if (this._challenger[idx] > this.hotHand.points + 6) {
        this.hotHand = { idx, points: this._challenger[idx], windowStart: now };
        this._challenger = {};
      }
    }
  }

  _trackRuns() {
    const su = this.api.getScoreUser(), sc = this.api.getScoreCPU();
    if (su > this.lastUserScore) {           // user scored
      this.userRun += (su - this.lastUserScore);
      this.cpuRun = 0;
    } else if (sc > this.lastCpuScore) {     // cpu scored
      this.cpuRun += (sc - this.lastCpuScore);
      this.userRun = 0;
    }
    // A defensive stop resets the bleeding slowly (prevents stale triggers).
    this.lastUserScore = su;
    this.lastCpuScore = sc;
  }

  /* ========================================================================
   * ADAPTIVE DEFENSE — the chess match
   * ====================================================================== */
  _updateDefense() {
    if (this.schemeCooldown > 0 || this.api.isPaused()) return;
    const p = this.patterns || this._recognizePatterns();
    const P = this.P;

    // Priority order: most exploitable pattern first.
    const candidates = [];

    // 1. Pick & roll spam -> hedge hard, then trap if it persists.
    if (p.pnrRate >= P.hedgeAfterPnR + P.adjustmentDelay) {
      candidates.push({
        scheme: p.pnrRate >= P.hedgeAfterPnR + 3 ? 'trap-pnr' : 'hedge',
        reason: `Hedging the pick & roll — you've run it ${p.pnrRate} times lately.`,
        weight: 3 + p.pnrRate,
      });
    }

    // 2. Three-point barrage -> zone or run-them-off.
    if (p.threeMakes >= P.zoneAfterThrees) {
      candidates.push({
        scheme: P.prefersZone ? 'zone-2-3' : 'chase',
        reason: `Adjusting to the three-ball — ${p.threeMakes} made threes in the window. ${P.prefersZone ? 'Dropping into zone.' : 'Running shooters off the line.'}`,
        weight: 3 + p.threeMakes,
      });
    }

    // 3. Star player cooking -> trap / box-and-one energy.
    if (p.userStarPts >= P.trapStarAfter && p.userStarIdx >= 0) {
      const name = this._userPlayerName(p.userStarIdx);
      candidates.push({
        scheme: 'trap-star',
        reason: `Sending help at ${name} — ${p.userStarPts} points won't go unanswered.`,
        weight: 2 + p.userStarPts / 6,
        starIdx: p.userStarIdx,
      });
    }

    // 4. Post-up spam -> double the post.
    if (p.paintTouches >= P.doublePostAfter + P.adjustmentDelay) {
      candidates.push({
        scheme: 'double-post',
        reason: `Doubling the post — ${p.paintTouches} paint touches is too many.`,
        weight: 2 + p.paintTouches,
      });
    }

    // 5. Isolation heavy -> load up strong side, invite the drive into help.
    if (p.isoRate >= 5) {
      candidates.push({
        scheme: 'shade-iso',
        reason: `Loading up on isolation — make someone else beat us.`,
        weight: 1.5 + p.isoRate / 2,
      });
    }

    if (!candidates.length) {
      // Nothing exploitable: drift back to base man, quietly.
      if (this.scheme !== 'man' && this._patternGoneCold()) {
        this._setDefensiveScheme('man', 'Back to base man — the wrinkle is handled.', true);
      }
      return;
    }

    candidates.sort((a, b) => b.weight - a.weight);
    const pick = candidates[0];
    if (pick.scheme === this.scheme) return;

    // Human imperfection: stubbornness — occasionally ignore the signal once.
    if (Math.random() < P.stubbornness && !pick.forced) {
      this._stubbornNote = pick;      // remember; will act next window
      this.schemeCooldown = 6;
      return;
    }
    // If we were stubborn last window about THIS, do it now (delayed reaction).
    const finalPick = (this._stubbornNote && this._stubbornNote.scheme === pick.scheme)
      ? { ...pick, reason: pick.reason + ' (Took a possession to get there.)' }
      : pick;
    this._stubbornNote = null;

    this._setDefensiveScheme(finalPick.scheme, finalPick.reason, false, finalPick);
  }

  _patternGoneCold() {
    // The exploited pattern hasn't shown up in the last several plays.
    const recent = this.playHistory.slice(-6);
    if (!recent.length) return true;
    const p = this.patterns || {};
    if (this.scheme === 'hedge' || this.scheme === 'trap-pnr')
      return !recent.some(r => r.play === 'pick-roll');
    if (this.scheme === 'zone-2-3' || this.scheme === 'chase')
      return (p.threeRate || 0) < 2;
    return recent.length >= 6;
  }

  _setDefensiveScheme(scheme, reason, silent = false, extra = {}) {
    this.scheme = scheme;
    this.schemeCooldown = 14; // ~2 possessions before another shift
    this.api.applyDefensiveScheme(scheme, { starIdx: extra.starIdx, silent: true });
    if (!silent) {
      const teamName = this.api.getCpuTeamName();
      this.api.announce(
        `${teamName.toUpperCase()} ADJUSTMENT`,
        reason,
        'defense'
      );
    }
  }

  _userPlayerName(idx) {
    try {
      const r = this.api.rosterFor('user', idx);
      return r.name || `Player ${idx + 1}`;
    } catch (e) { return `Player ${idx + 1}`; }
  }

  /* ========================================================================
   * TIMEOUT INTELLIGENCE
   * ====================================================================== */
  _manageTimeouts() {
    if (this.timeoutCooldown > 0 || this.timeoutsLeft <= 0) return;
    if (this.api.isPaused() || this.api.isGameOver()) return;
    const diff = this.api.getScoreCPU() - this.api.getScoreUser(); // + means CPU leads

    // 1. Stop the bleeding: user on a run.
    if (this.userRun >= this.P.runTimeoutAt && diff < 12) {
      return this._callTimeout(
        'STOP THE RUN',
        `User run at ${this.userRun}-0. ${this.P.coachName} wants to talk it over.`
      );
    }

    // 2. Before the user's final possession (close game, user ball, <30s).
    const quarter = this.api.getGameQuarter();
    const qRem = this._qRemaining();
    if (quarter === 4 && qRem < 32 && qRem > 4 &&
        Math.abs(diff) <= 6 && this.api.getBallTeam() === 'user' &&
        !this._finalPossessionTOUsed) {
      this._finalPossessionTOUsed = true;
      return this._callTimeout(
        'SET THE DEFENSE',
        'Final possession coming — getting the right five on the floor.'
      );
    }

    // 3. Ice the free-throw shooter (clutch FTs, game on the line).
    if (this.api.isUserShootingFT() && quarter === 4 && qRem < 60 &&
        Math.abs(diff) <= 4 && Math.random() < 0.6) {
      return this._callTimeout('ICE', 'Icing the shooter. Let him think about it.');
    }

    // 4. Protect a late lead: slow the game down, sub defense in.
    if (quarter === 4 && qRem < 90 && diff >= 8 && diff <= 14) {
      return this._callTimeout(
        'CLOSE IT OUT',
        'Up ' + diff + ' — shortening the game, no silly fouls.'
      );
    }

    // 5. Never waste: don't call when CPU is rolling or game is decided.
    // (Implicit — none of the above fired.)
  }

  _callTimeout(title, detail) {
    this.timeoutsLeft--;
    this.timeoutCooldown = 20;
    this.userRun = 0; // reset run tracking after the stoppage
    this.api.cpuCallTimeout(title, detail, this.timeoutsLeft);
  }

  /* ========================================================================
   * ROTATION MANAGEMENT
   * ====================================================================== */
  _manageRotation(dt) {
    if (this.subCooldown > 0 || this.api.isPaused()) return;
    const cpu = this.api.getCpuTeam();
    if (!cpu || !cpu.length) return;

    for (let i = 0; i < cpu.length; i++) {
      const p = cpu[i].userData;
      if (!p || p.fouledOut) continue;

      // 1. Fatigue: legs are gone.
      if (this.api.isFatigueOn() && p.stamina < this.P.fatigueSubAt && this._benchAvailable()) {
        return this._substitute(i, `${this._cpuPlayerName(i)} is gassed (${Math.round(p.stamina)}%). Fresh legs in.`);
      }

      // 2. Foul trouble: protect the starter.
      const foulLimit = this.api.getGameQuarter() <= 2 ? this.P.foulTroubleAt : this.P.foulTroubleAt + 1;
      if (p.fouls >= foulLimit && this._benchAvailable()) {
        return this._substitute(i, `${this._cpuPlayerName(i)} has ${p.fouls} fouls — protecting him for the finish.`);
      }

      // 3. Matchup: user's star is killing us — get a stopper in.
      if (this._userStarIdx >= 0 && this._userStarPts >= 14 && i === this._cpuWorstDefenderVs(this._userStarIdx)) {
        const stopper = this._findDefensiveStopper();
        if (stopper >= 0 && stopper !== i) {
          return this._substitute(i, `Getting a stopper in on ${this._userPlayerName(this._userStarIdx)}.`);
        }
      }
    }

    // 4. Ride the hot hand: extend the hot player's run (skip their normal rest).
    if (this.hotHand.idx >= 0 && this.hotHand.points >= 12) {
      this._hotHandProtected = this.hotHand.idx;
    }
  }

  _benchAvailable() {
    try {
      const bench = this.api.getCpuBench();
      return bench && bench.length > 0 && this.api.getCpuSubsLeft() > 0;
    } catch (e) { return false; }
  }

  _substitute(outIdx, reason) {
    if (this._hotHandProtected === outIdx) return; // don't pull the hot hand
    const ok = this.api.cpuSubstitute(outIdx, reason);
    if (ok) {
      this.subCooldown = 25;
      this.api.announce('CPU SUBSTITUTION', reason, 'rotation');
    }
  }

  _cpuPlayerName(idx) {
    try { return this.api.rosterFor('cpu', idx).name || `CPU ${idx + 1}`; }
    catch (e) { return `CPU ${idx + 1}`; }
  }

  _cpuWorstDefenderVs(userStarIdx) {
    // Crude: the CPU defender most recently scored on by the star.
    // Host can override via api.getCpuDefensiveMatchup(starIdx).
    try {
      if (this.api.getCpuDefensiveMatchup) return this.api.getCpuDefensiveMatchup(userStarIdx);
    } catch (e) {}
    return Math.floor(Math.random() * 5);
  }

  _findDefensiveStopper() {
    try {
      const bench = this.api.getCpuBench() || [];
      // Bench entries: [name, number, pos, rating]. Prefer wings/big for stops.
      let best = -1, bestScore = -1;
      bench.forEach((b, i) => {
        const pos = b[2] || '';
        const score = (b[3] || 60) + (/SF|PF|SG/.test(pos) ? 8 : 0);
        if (score > bestScore) { bestScore = score; best = i; }
      });
      return best;
    } catch (e) { return -1; }
  }

  /* ========================================================================
   * LATE-GAME STRATEGY
   * ====================================================================== */
  _manageLateGame() {
    if (this.api.isPaused() || this.api.isGameOver()) return;
    const qRem = this._qRemaining();
    if (this.api.getGameQuarter() !== 4 || qRem > 120) {
      this.foulStrategyArmed = false;
      return;
    }
    const diff = this.api.getScoreCPU() - this.api.getScoreUser();

    // 1. Down 3+ with <30s and user has the ball: FOUL. Target the poor FT shooter.
    if (diff <= -3 && qRem < 30 && this.api.getBallTeam() === 'user' && !this.foulStrategyArmed) {
      const target = this._findWorstFTShooter('user');
      this.foulStrategyArmed = true;
      this.api.setFoulStrategy({ active: true, targetIdx: target });
      this.api.announce(
        'FOUL TO EXTEND',
        `Down ${-diff} — fouling ${this._userPlayerName(target)} (${this._userFT(target)}% FT) to extend the game.`
      );
      return;
    }
    if (this.foulStrategyArmed && (diff > -3 || this.api.getBallTeam() !== 'user')) {
      this.foulStrategyArmed = false;
      this.api.setFoulStrategy({ active: false });
    }

    // 2. Up 3, under 15s, user has ball: foul BEFORE the three (no and-one prayer).
    //    Conservative coaches do this; aggressive ones trust the contest.
    if (diff === 3 && qRem < 15 && this.api.getBallTeam() === 'user' &&
        this.personality === 'grinder' && !this._fouledUpThree) {
      this._fouledUpThree = true;
      const target = this._findWorstFTShooter('user');
      this.api.setFoulStrategy({ active: true, targetIdx: target, upThree: true });
      this.api.announce('FOUL UP THREE', 'Not giving up the tying three. Make them earn it at the line.');
    }

    // 3. Tie or down 1-2, <24s, CPU ball: hold for the last shot.
    //    (Handled in _considerLastShotHold via possession start.)
  }

  _considerLastShotHold(team) {
    const qRem = this._qRemaining();
    const diff = this.api.getScoreCPU() - this.api.getScoreUser();
    if (this.api.getGameQuarter() === 4 && team === 'cpu' &&
        qRem < 30 && qRem > 8 && diff >= -2 && diff <= 2 &&
        !this.lastShotHeld) {
      this.lastShotHeld = true;
      this.api.setCpuTempo('stall');
      this.api.announce('LAST SHOT', 'Holding for the final look. No rush — one good shot wins it.');
    }
    if (this.lastShotHeld && (qRem <= 8 || team !== 'cpu')) {
      this.lastShotHeld = false;
      this.api.setCpuTempo(this.P.pace); // restore
    }
  }

  _findWorstFTShooter(team) {
    let worst = 0, worstFT = 101;
    for (let i = 0; i < 5; i++) {
      try {
        const ft = this._userFT(i);
        // Skip the ball handler if they're elite? No — hack-a-whoever is weakest.
        if (ft < worstFT) { worstFT = ft; worst = i; }
      } catch (e) {}
    }
    return worst;
  }

  _userFT(idx) {
    try {
      const r = this.api.rosterFor('user', idx);
      return r.ft || r[3] || 70;
    } catch (e) { return 70; }
  }

  /* ========================================================================
   * HALFTIME ADJUSTMENTS
   * ====================================================================== */
  _applyHalftimeAdjustments() {
    const h = this.half;
    const adjustments = [];

    // 1. Perimeter: user burned us from three.
    const threePct = h.userThreesAttempted > 0 ? h.userThreesMade / h.userThreesAttempted : 0;
    if (h.userThreesAttempted >= 8 && threePct >= 0.38) {
      adjustments.push({
        scheme: this.P.prefersZone ? 'zone-2-3' : 'chase',
        text: `They shot ${(threePct * 100).toFixed(0)}% from three (${h.userThreesMade}/${h.userThreesAttempted}). ${this.P.prefersZone ? 'Zone it up, close out short.' : 'No more open looks — chase everything.'}`,
      });
    } else if (h.userThreesAttempted >= 10 && threePct < 0.28) {
      adjustments.push({
        scheme: 'pack',
        text: `They're ${(threePct * 100).toFixed(0)}% from deep — pack it in, dare them to keep missing.`,
      });
    }

    // 2. Paint: user lived in the lane.
    if (h.userPaintPoints >= 20) {
      adjustments.push({
        scheme: 'protect-paint',
        text: `${h.userPaintPoints} paint points in the half. Wall off the lane.`,
      });
    }

    // 3. Pick & roll: staple of their half.
    if (h.userPnR >= 6) {
      adjustments.push({
        scheme: 'hedge',
        text: `${h.userPnR} pick-and-rolls — we're hedging everything coming out of the break.`,
      });
    }

    // 4. Turnovers: if we're forcing them, press more.
    if (h.userTurnovers >= 6 && this.personality === 'blur') {
      adjustments.push({
        scheme: 'press',
        text: `${h.userTurnovers} turnovers forced — turning up the pressure.`,
      });
    }

    // 5. Star: their best player.
    if (this._userStarIdx >= 0 && this._userStarPts >= 12) {
      adjustments.push({
        scheme: 'trap-star',
        starIdx: this._userStarIdx,
        text: `${this._userPlayerName(this._userStarIdx)} has ${this._userStarPts}. He's seeing two jerseys in the second half.`,
      });
    }

    if (!adjustments.length) {
      this.api.announce('HALFTIME', `${this.P.coachName}: "Stay the course. First to blink loses." No major changes.`, 'halftime');
      return;
    }

    // Apply the highest-priority adjustment as the base second-half scheme.
    const primary = adjustments[0];
    this._setDefensiveScheme(primary.scheme, primary.text, true, primary);
    this.api.announce(
      'HALFTIME ADJUSTMENTS',
      adjustments.map(a => '• ' + a.text).join('\n'),
      'halftime'
    );
  }

  /* ========================================================================
   * UTILITIES
   * ====================================================================== */
  _qRemaining() {
    // Seconds REMAINING in the current quarter.
    const t = this.api.getGameSeconds();
    const ql = this.api.getQuarterLength();
    return ql - (t % ql);
  }

  resetForNewGame() {
    const fresh = new CPUCoach(this.api, { personality: this.personality });
    Object.assign(this, fresh);
  }
}

/* Zone / press / trap positioning. Called from teamAI's defense loop
   (integration step 4). Returns a Vector3 target or null (use default). */
function coachDefenderTarget(p,i,offense,mark,ballP,dir){
  if(!coachMode||ballTeam!=='cpu') return null;      // schemes apply when WE defend
  const s=COACH_SCHEMES[coachScheme]; if(!s) return null;
  const bx=basketFor('user').pos.x, cx=(-Math.sign(bx))||1;  // toward halfcourt
  const cl=(x,z)=>V(THREE.MathUtils.clamp(x,-13,13),0,THREE.MathUtils.clamp(z,-6.4,6.4));
  if(s.zone){
    const bz=ballP.z*.18;
    const spots = s.zone==='23'
      ? [[bx+cx*4.6,-2.6+bz],[bx+cx*4.6,2.6+bz],[bx+cx*1.4,-4.6],[bx+cx*1.0,bz],[bx+cx*1.4,4.6]]
      : [[bx+cx*4.2,-4.2],[bx+cx*4.8,bz],[bx+cx*4.2,4.2],[bx+cx*1.2,-2.6],[bx+cx*1.2,2.6]];
    const sp=spots[i%spots.length];
    return cl(sp[0],sp[1]);
  }
  if(s.press){
    const order=playerTeam.map((q,j)=>({j,d:q.position.distanceTo(ballP)})).sort((a,b)=>a.d-b.d);
    if(order[0].j===i) return ballP.clone();                                   // on the ball, full court
    if(order[1]&&order[1].j===i) return cl(ballP.x+cx*1.4, ballP.z+(i%2?1.4:-1.4));
    return null;                                                               // rest: default man
  }
  if(s.trap && Math.sign(ballP.x)===Math.sign(bx)){
    const order=playerTeam.map((q,j)=>({j,d:q.position.distanceTo(ballP)})).sort((a,b)=>a.d-b.d);
    if(order[0].j===i||(order[1]&&order[1].j===i)) return cl(ballP.x,ballP.z);   // the trap
    const back=[[bx+cx*1.6,-4.4],[bx+cx*1.2,0],[bx+cx*1.6,4.4]];                // rotate behind
    const k=[0,1,2,3,4].filter(j=>j!==order[0].j&&j!==(order[1]&&order[1].j)).indexOf(i);
    const sp=back[Math.max(0,k)%3];
    return cl(sp[0],sp[1]);
  }
  return null;
}

/* Watches score flow and game state; returns a timeout suggestion or null. */
function coachTimeoutAdvisor(){
  if(!coachMode||!run||paused||gameEnded) return null;
  if(coachTimeouts<=0||coachTimeoutLock>0) return null;
  const diff=scoreUser-scoreCPU;
  const log=coachScoreLog;
  if(log.length>2){
    const cutoff=gameSeconds+150;
    const old=log.find(e=>e.t>=cutoff)||log[0];
    const slide=old.diff-diff;                       // positive = opponent run
    if(slide>=8) return {label:'STOP THE RUN', reason:`Opponent outscoring you by ${slide} lately. Break their rhythm.`};
  }
  if(gameQuarter===4&&gameSeconds<=35&&Math.abs(diff)<=6)
    return {label:'SET FINAL SHOT', reason:'Draw up the last look of the quarter.'};
  if(gameQuarter>=3){
    const gassed=playerTeam.findIndex(p=>!p.userData.fouledOut&&p.userData.stamina<28);
    if(gassed>=0) return {label:'REST STARTERS', reason:`${rosterFor('user',gassed).name} is gassed (${Math.round(playerTeam[gassed].userData.stamina)}%).`};
  }
  return null;
}

/* Late-game situation board. */
function coachLateAdvisor(){
  if(!coachMode||!run||gameEnded||gameQuarter!==4) return null;
  const diff=scoreUser-scoreCPU;
  if(gameSeconds<75&&diff<0&&diff>=-8&&ballTeam==='cpu'&&!shot)
    return {label:'FOUL NOW', reason:'Trailing — stop the clock and extend the game.', action:'foul'};
  if(gameSeconds<75&&diff>0&&ballTeam==='cpu'&&!shot)
    return {label:'NO FOUL', reason:'Protect the lead. Make them beat you 5-on-5.'};
  if(gameSeconds<=26&&ballTeam==='user'&&Math.abs(diff)<=5&&!shot)
    return {label:'LAST SHOT', reason:'Timeout now to set up the final look.', action:'timeout'};
  if(gameSeconds<=12&&diff<=-1&&diff>=-3&&ballTeam==='user'&&!shot)
    return {label:'NEED THREE', reason:'Run HORNS or PERIMETER action for the tie.', action:'play', play:'horns'};
  return null;
}

/* Targeted substitution: bench player bi replaces court player ci. */
function coachSubstitution(ci,bi){
  if(!coachMode||!run||paused||gameEnded) return;
  if(coachSubCount<=0){ coachSetReport('No subs left','The rotation is locked for the night.'); return; }
  const entry=coachBenchEntry(bi);
  if(!entry){ coachSetReport('No bench data','This roster has no bench configured.'); return; }
  const model=playerTeam[ci];
  if(!model||model.userData.fouledOut){ coachSetReport('Cannot sub','Pick a player who is on the floor.'); return; }
  const previous=rosterFor('user',ci).name;
  model.userData.roster={name:entry[0],number:entry[1],pos:entry[2],ft:entry[3]};
  model.userData.stamina=100; model.userData.fouls=0;
  const draw=(canvas,fn)=>{const ctx=canvas.getContext('2d');ctx.clearRect(0,0,canvas.width,canvas.height);fn(ctx);};
  draw(model.userData.numberCanvas,ctx=>{ctx.fillStyle='#f4f4e9';ctx.font='900 90px Arial';ctx.textAlign='center';ctx.textBaseline='middle';ctx.fillText(String(entry[1]),64,65);});
  draw(model.userData.nameCanvas,ctx=>{ctx.fillStyle='#f4f4e9';ctx.font='900 26px Arial';ctx.textAlign='center';ctx.textBaseline='middle';ctx.fillText(String(entry[0]).toUpperCase(),128,24);});
  model.userData.numberTexture.needsUpdate=true;
  model.userData.nameTexture.needsUpdate=true;
  active=ci;                                   // keep indicator on the slot; ball stays put
  coachSubCount--;
  const s1=$('coachSubs'); if(s1) s1.textContent=`SUBS ${coachSubCount}/5`;
  coachSelectedCourt=-1;
  coachSetReport('Substitution',`${previous} sits. ${entry[0]} (${entry[2]}) checks in with fresh legs.`);
  CoachClipboard.renderRotation(); CoachClipboard.sync();
}

/* Quick sub: best bench option for the most tired (or fouled) starter. */
function coachQuickSub(){
  let ci=-1, score=-1;
  for(let i=0;i<5;i++){ const p=playerTeam[i];
    if(p.userData.fouledOut) continue;
    const s=(100-p.userData.stamina)+(p.userData.fouls>=4?30:0);
    if(s>score){score=s;ci=i;} }
  if(ci<0||score<12){ coachSetReport('Rotation is fine','Nobody needs a breather right now.'); return; }
  const bench=(chosenTeam('user').bench||[]);
  let bi=0, br=-1;
  bench.forEach((e,j)=>{ const r=e[3]||70; if(r>br){br=r;bi=j;} });
  coachSubstitution(ci,bi);
}

function coachFeedCloser(){

  playerTeam.forEach((p,i)=>p.userData.hasBall=(i===c));
  ballHandler=c; active=c;
  coachSetReport('Feed the closer',`${rosterFor('user',c).name} has the ball with the game on the line.`);
  return true;
}
/* Clutch auto-feed: called from onPossession when the game is tight late. */
function coachClutchCheck(){
  if(gameQuarter!==4||gameSeconds>120) return;
  if(Math.abs(scoreUser-scoreCPU)>8) return;
  if(ballTeam==='user'&&!shot&&!passFlight&&ballHandler!==coachCloser&&Math.random()<.5)
    coachFeedCloser();
}

function coachFoulNow(){
  if(!coachMode||!run||paused||gameEnded) return;
  const ok=gameQuarter===4&&gameSeconds<75&&scoreUser<scoreCPU&&ballTeam==='cpu'&&!shot&&!passFlight;
  if(!ok){ coachSetReport('No foul needed','Intentional fouls are for trailing late with them holding the ball.'); return; }
  const h=getHandler();
  let best=0,bd=1e9;
  playerTeam.forEach((p,i)=>{ if(p.userData.fouledOut) return;
    const d=p.position.distanceTo(h.position); if(d<bd){bd=d;best=i;} });
  gameSeconds=Math.max(0,gameSeconds-4);          // free throws eat clock
  resolveShootingFoul({foul:true,team:'cpu',handler:ballHandler,foulerIndex:best,points:2},false);
  switchPossession('cpu');                        // they keep the ball after FTs
  const fouler=rosterFor('user',best);
  const foulCount=playerTeam[best].userData.fouls||0;
  coachSetReport('Intentional foul',`${fouler.name} (${foulCount} fouls) stops the clock. They shoot two.`);
  updateUI(); CoachClipboard.sync();
}

/* ---- gameAPI adapter for CPUCoach ---- */
function cpuGameAPI(){
  return {
    getScoreUser: () => scoreUser,
    getScoreCPU: () => scoreCPU,
    getGameSeconds: () => (settings.quarterMinutes*60*4) - gameSeconds,
    getQuarterLength: () => settings.quarterMinutes*60,
    getGameQuarter: () => gameQuarter,
    getBallTeam: () => ballTeam,
    getShotSeconds: () => shotSeconds,
    getPlayerTeam: () => playerTeam,
    getCpuTeam: () => cpuTeam,
    getCpuBench: () => (chosenTeam('cpu').bench || []),
    getCpuSubsLeft: () => cpuSubsLeft,
    getCpuTeamId: () => ($('awayTeam') ? $('awayTeam').value : 'cpu'),
    getCpuTeamName: () => chosenTeam('cpu').name,
    rosterFor: (s,i) => rosterFor(s,i),
    isPaused: () => paused,
    isGameOver: () => gameEnded,
    isFatigueOn: () => !!settings.fatigue,
    isUserShootingFT: () => !!freeThrowMode && freeThrowMode.team === 'user',
    applyDefensiveScheme: (scheme, opts) => {
      const map = { man:65, hedge:80, 'trap-pnr':88, 'trap-star':84,
        'zone-2-3':78, chase:60, 'double-post':85, 'shade-iso':75,
        'protect-paint':90, pack:95, press:70 };
      settings.helpDefense = (map[scheme] !== undefined) ? map[scheme] : 65;
      doubleTeamMode = /trap|double/.test(scheme);
      cpuDefScheme = scheme;
    },
    setCpuTempo: (t) => { cpuTempo = t; },
    setFoulStrategy: (s) => { cpuFoulPlan = s || { active:false }; },
    cpuCallTimeout: (title, detail, left) => {
      if (paused || gameEnded || !run) return;
      cpuTimeouts = Math.max(0, left);
      shot = null; passFlight = null; shotSeconds = 24; possessionSeconds = 0;
      paused = true;
      $('pause').classList.remove('hidden');
      $('pauseText').textContent = `CPU TIMEOUT · ${title} · ${scoreUser}–${scoreCPU} · ${Math.floor(gameSeconds/60)}:${String(Math.floor(gameSeconds%60)).padStart(2,'0')} remaining.`;
      $('resume').textContent = 'RESUME GAME';
      showToast('CPU TIMEOUT · ' + title, detail, 1600);
      coachOnTimeout('cpu');
    },
    cpuSubstitute: (outIdx, reason) => {
      const bench = chosenTeam('cpu').bench || [];
      if (!bench.length || cpuSubsLeft <= 0) return false;
      const model = cpuTeam[outIdx];
      if (!model || model.userData.fouledOut) return false;
      const entry = bench[Math.floor(Math.random()*bench.length)];
      const previous = model.userData.roster ? model.userData.roster.name : rosterFor('cpu', outIdx).name;
      model.userData.roster = { name: entry[0], number: entry[1], pos: entry[2], ft: entry[3] };
      try {
        const draw = (canvas, fn) => { const ctx = canvas.getContext('2d'); ctx.clearRect(0,0,canvas.width,canvas.height); fn(ctx); };
        draw(model.userData.numberCanvas, ctx => { ctx.fillStyle='#f4f4e9'; ctx.font='900 90px Arial'; ctx.textAlign='center'; ctx.textBaseline='middle'; ctx.fillText(String(entry[1]),64,65); });
        model.userData.numberTexture.needsUpdate = true;
        draw(model.userData.nameCanvas, ctx => { ctx.fillStyle='#f4f4e9'; ctx.font='900 26px Arial'; ctx.textAlign='center'; ctx.textBaseline='middle'; ctx.fillText(entry[0].toUpperCase(),128,24); });
        model.userData.nameTexture.needsUpdate = true;
      } catch(e){}
      cpuSubsLeft--;
      try {
        Commentary.onSub(entry[0], previous, chosenTeam('cpu').name);
        const lineup = cpuTeam.map((p,i)=>p.userData.roster?p.userData.roster.name:rosterFor('cpu',i).name);
        CourtsideStats.setLineup('away', lineup);
      } catch(e){}
      return true;
    },
    announce: (title, detail, kind) => {
      try { showToast(title, detail, 1500); } catch(e){}
      try { coachSetReport(title, detail); } catch(e){}
    }
  };
}

/* ---- shot resolution fan-out ---- */
function coachOnShotResolved(s, made, team, pts, typ){
  if (!coachMode) return;
  try {
    const hIdx = s.handler || 0;
    const r = rosterFor(team, hIdx);
    const name = r.name, abbr = chosenTeam(team).abbr, teamName = chosenTeam(team).name;
    const cmTeam = (team === 'user') ? 'home' : 'away';
    const clutch = (gameQuarter >= 4 && gameSeconds < 120 && Math.abs(scoreUser - scoreCPU) <= 8);
    const atBuzzer = gameSeconds <= 1.6;
    const ctx = { clutch: clutch, atBuzzer: atBuzzer };
    if (window.Momentum && made) Momentum.onScore(team, pts, typ);
    if (window.Commentary) {
      if (atBuzzer && made) Commentary.onBuzzerBeater(name, pts, true, ctx);
      else if (made && pts === 3) Commentary.onThreeMade(name, abbr, teamName, ctx);
      else if (made && typ === 'DUNK') Commentary.onDunk(name, abbr, teamName, ctx);
      else if (made && typ === 'LAYUP') Commentary.onLayup(name, abbr, teamName, ctx);
      else if (made) Commentary.onTwoMade(name, abbr, teamName, ctx);
      else if (clutch) Commentary.onClutchShot(name, false, ctx);
      else Commentary.onMiss(name, abbr, teamName, pts === 3, ctx);
    }
    if (window.CourtsideStats) {
      if (made) CourtsideStats.score({ team: cmTeam, player: name, points: pts, kind: typ });
      else CourtsideStats.miss({ team: cmTeam, player: name, points: pts, kind: typ });
    }
    if (window.PlayVisuals && s.start) {
      try { PlayVisuals.recordShot(s.start.x, s.start.z, made, team, hIdx); } catch(e){}
    }
