/* ---------------- MOMENTUM CORE ---------------- */

/** Called on every made basket. team: 'user'|'cpu', pts: 2|3, kind: 'DUNK'|'LAYUP'|'JUMPER' etc. */
function onScore(team, pts, kind) {
  const dir = team === 'user' ? 1 : -1;
  let shift = pts * 2.2;                       // 2pt = 4.4, 3pt = 6.6
  if (kind === 'DUNK') shift += 4;             // dunks electrify
  if (pts === 3) shift += 2;                   // threes swing momentum
  // Diminishing returns if already heavily favored
  if (Math.sign(M.value) === dir) shift *= Math.max(.35, 1 - Math.abs(M.value) / 140);
  M.value = clamp(M.value + dir * shift, -100, 100);

  // Log for run detection (use game clock)
  const t = G('gameSeconds', 0);
  M.scoreLog.push({ team, pts, t });
  if (M.scoreLog.length > 40) M.scoreLog.shift();

  // Crowd: home crowd erupts for user scores, deflates for CPU scores
  bumpCrowd(dir * (4 + pts * 1.5));

  detectRun();
  updateWinProb();
  renderMeter();
}

/** Called on blocks, steals, charges drawn. team: who made the play. */
function onBigPlay(team, playType) {
  const dir = team === 'user' ? 1 : -1;
  const amount = playType === 'BLOCK' ? 5 : playType === 'STEAL' ? 4 : 3;
  M.value = clamp(M.value + dir * amount, -100, 100);
  bumpCrowd(dir * 5);
  const label = { BLOCK: 'REJECTED!', STEAL: 'PICKED CLEAN!', CHARGE: 'CHARGE DRAWN!' }[playType] || 'BIG PLAY!';
  const who = team === 'user' ? 'YOUR DEFENSE' : 'CPU DEFENSE';
  callGame('showToast', label, `${who} swings the momentum.`, 900);
  renderMeter();
}

/** Called at the end of coachCallTimeout(). */
function onTimeout() {
  // If the opponent is on a run, timeout has a 65% chance to kill it
  if (M.runActive && M.runActive.team === 'cpu') {
    if (Math.random() < 0.65) {
      const killed = M.runActive.points;
      M.value = clamp(M.value * 0.35, -100, 100);  // bleed off CPU momentum
      M.runActive = null;
      M.runAlerted = '';
      M.scoreLog.length = 0;
      bumpCrowd(12);  // crowd settles, coach's message lands
      callGame('showToast', 'BLEEDING STOPPED', `Timeout kills the ${killed}-0 run. Regroup.`, 1400);
      callGame('coachSetReport', 'Run stopped', 'The timeout broke their rhythm. Reset the offense.');
    } else {
      callGame('showToast', 'RUN CONTINUES', 'The timeout didn\'t cool them off. Dig deeper.', 1400);
      callGame('coachSetReport', 'Still their run', 'Timeout didn\'t break it. Consider a substitution or defensive change.');
    }
  } else if (M.runActive && M.runActive.team === 'user') {
    // Calling timeout on your OWN run is usually a mistake — small penalty
    M.value = clamp(M.value * 0.85, -100, 100);
    callGame('coachSetReport', 'Timeout on your run', 'Careful — you may have cooled off your own hot hand.');
  } else {
    // Neutral timeout: slight reset toward center, crowd catches breath
    M.value *= 0.7;
    bumpCrowd(-6);
  }
  renderMeter();
}

/* ---------------- RUN DETECTION ---------------- */
function detectRun() {
  // Walk back through the log: count consecutive unanswered points.
  // A run is live when one team has 8+ unanswered, whether the run
  // has ended (opponent finally scored) or is still going.
  let team = null, points = 0;
  for (let i = M.scoreLog.length - 1; i >= 0; i--) {
    const s = M.scoreLog[i];
    if (team === null) { team = s.team; points = s.pts; }
    else if (s.team === team) { points += s.pts; }
    else break; // opponent answered — run boundary found
  }
  if (team === null || points < 8) { M.runActive = null; return; }

  const key = `${team}-${points}`;
  M.runActive = { team, points, key };
  if (M.runAlerted === key) return;  // already showed this run
  M.runAlerted = key;

  const who = team === 'user' ? 'YOUR TEAM' : 'CPU';
  const score = `${G('scoreUser', 0)}–${G('scoreCPU', 0)}`;
  showRunGraphic(points, who);

  if (team === 'cpu') {
    // Suggest timeout — this is the coaching moment
    callGame('showToast', `${points}-0 CPU RUN`, 'Consider calling a timeout to stop the bleeding.', 1800);
    callGame('coachSetReport', 'Opponent run!', `${points}-0 run against you. TIMEOUT is available — use it to break their rhythm.`);
    callGame('setState', 'OPPONENT RUN', `${points} unanswered. The building is quiet. Call timeout?`);
  } else {
    callGame('showToast', `${points}-0 RUN!`, 'Your team is rolling. The crowd is electric.', 1500);
    callGame('setState', 'YOUR RUN', `${points} unanswered points. Keep attacking.`);
  }
}

function showRunGraphic(points, who) {
  const rg = M.els.runGraphic;
  if (!rg) return;
  rg.querySelector('.rg-main').textContent = `${points}-0`;
  rg.querySelector('.rg-sub').textContent = `${who} · ${G('scoreUser', 0)}–${G('scoreCPU', 0)}`;
  rg.classList.add('show');
  clearTimeout(M.toastTimer);
  M.toastTimer = setTimeout(() => rg.classList.remove('show'), 2600);
}

/* ---------------- CROWD ENERGY ---------------- */
function bumpCrowd(amount) {
  M.crowdEnergy = clamp(M.crowdEnergy + amount, 0, 100);
  renderCrowd();
}

/** Multiplier for crowd audio gain. Call where crowdGain is set. */
function getCrowdVolumeMult() {
  return 0.35 + (M.crowdEnergy / 100) * 1.3;  // 0.35x .. 1.65x
}

function renderCrowd() {
  const bars = M.els.crowdBars;
  if (!bars || !bars.length) return;
  const lit = Math.round((M.crowdEnergy / 100) * bars.length);
  bars.forEach((b, i) => {
    const on = i < lit;
    b.style.height = on ? `${8 + (i / bars.length) * 18}px` : '6px';
    b.style.background = !on ? '#2a3542'
      : M.crowdEnergy > 75 ? '#d8ff48'
      : M.crowdEnergy > 45 ? '#f3b544' : '#7a6a4a';
  });
}

/* ---------------- SHOOTING IMPACT ---------------- */
/** Returns a multiplier for shot chance. Call at each shot computation. */
function getShootingMult(team) {
  const m = team === 'user' ? M.value : -M.value;  // positive = this team has momentum
  let mult = 1 + (m / 100) * 0.06;                  // ±6% max
  // Clutch tightens everyone up slightly, but hot teams rise
  if (M.clutchOn) {
    mult *= 0.97;
    if (m > 30) mult *= 1.05;   // clutch gene: hot players want the moment
  }
  return mult;
}

/* ---------------- CLUTCH ---------------- */
function checkClutch() {
  const secs = G('gameSeconds', 999);
  const q = G('gameQuarter', 1);
  const qLen = 180; // default 3-min quarters; clutch = last 2 min of 4th
  const diff = Math.abs(G('scoreUser', 0) - G('scoreCPU', 0));
  const was = M.clutchOn;
  M.clutchOn = (q >= 4 && secs < 120 && diff <= 8 && !G('practice', false) && !G('gameEnded', false));
  if (M.clutchOn && !was) {
    M.els.clutch && M.els.clutch.classList.add('show');
    callGame('showToast', 'CLUTCH TIME', 'Final 2 minutes. Every possession matters.', 1600);
    bumpCrowd(18);
  } else if (!M.clutchOn && was) {
    M.els.clutch && M.els.clutch.classList.remove('show');
  }
}

/* ---------------- WIN PROBABILITY & COMEBACK ---------------- */
function updateWinProb() {
  const du = G('scoreUser', 0), dc = G('scoreCPU', 0);
  const diff = du - dc;                       // positive = user leads
  const secs = G('gameSeconds', 240);
  const total = 720;                          // 4 x 3-min quarters default
  const timeLeft = clamp(secs / total, 0, 1); // 1 = whole game left

  // Track extremes for comeback detection
  if (diff < 0) M.largestDeficit = Math.max(M.largestDeficit, -diff);
  if (diff > 0) M.largestLead = Math.max(M.largestLead, diff);

  // Simple logistic-ish model: lead matters more late
  const leverage = 1.6 - timeLeft;            // 0.6 early .. 1.6 late
  let p = 50 + diff * 3.2 * leverage + (M.value / 100) * 6;
  M.winProb = clamp(Math.round(p), 1, 99);

  // Comeback alert: was down 15+, now within 5
  if (!M.comebackAlerted && M.largestDeficit >= 15 && diff >= -5 && diff < 0) {
    M.comebackAlerted = true;
    showRunGraphic('COMEBACK', 'FROM DOWN ' + M.largestDeficit);
    callGame('showToast', 'COMEBACK BID', `Erased a ${M.largestDeficit}-point deficit. One stop at a time.`, 1800);
    bumpCrowd(25);
  }
  // Blowout watch: up 20+ late
  if (diff >= 20 && timeLeft < 0.25 && !M._blowout) {
    M._blowout = true;
    callGame('setState', 'BLOWOUT', 'This one is getting out of hand.');
  }
  renderWP();
}

function renderWP() {
  if (!M.els.wpFill) return;
  M.els.wpFill.style.width = M.winProb + '%';
  M.els.wpFill.style.background = M.winProb >= 50
    ? 'linear-gradient(90deg,#1d7a4c,#d8ff48)' : 'linear-gradient(90deg,#8c1f3f,#ff5a5a)';
  M.els.wpText.innerHTML = `${M.winProb}% <small>YOU</small>`;
}

function renderMeter() {
  if (!M.els.fillUser) return;
  const v = M.value; // -100..100
  M.els.fillUser.style.width = (v > 0 ? v / 2 : 0) + '%';
  M.els.fillCpu.style.width = (v < 0 ? -v / 2 : 0) + '%';
  M.els.momUser.classList.toggle('hot', v > 25);
  M.els.momCpu.classList.toggle('hot', v < -25);
}

/* ---------------- SIM CONTROLS ---------------- */
/**
 * Drop-in replacement for the per-frame updateGame(dt) call.
 * Handles 1x/2x/4x plus sim-to-whistle and sim-to-quarter.
 * Stoppages that end a sim: quarter change, game end, or (for
 * 'stoppage') any pause event (timeout). We detect pause via the
 * game's `paused` global flipping true.
 */
let _simIters = 0;
function tickSim(dt) {
  const updateGame = (() => { try { return eval('updateGame'); } catch (e) { return null; } })();
  if (typeof updateGame !== 'function') return;

  if (M.simTarget) {
    // Fast-forward in a tight loop with a safety cap
    const maxIters = 4000;
    let iters = 0;
    const startQ = G('gameQuarter', 1);
    const wasPaused = !!G('paused', false);
    while (iters++ < maxIters) {
      if (G('gameEnded', false)) break;
      if (G('gameQuarter', 1) !== startQ && M.simTarget === 'quarter') break;
      if (M.simTarget === 'stoppage' && !!G('paused', false) && !wasPaused) break;
      if (M.simTarget === 'stoppage' && G('gameQuarter', 1) !== startQ) break;
      updateGame(1 / 30);           // fixed 30fps step for stability
      Momentum.update(1 / 30);
      _simIters++;
    }
    M.simTarget = null;
    // reset sim buttons UI
    document.querySelectorAll('#simPanel button').forEach(x => x.classList.remove('active'));
    const b1 = document.querySelector('#simPanel button[data-sim="1"]');
    if (b1) b1.classList.add('active');
    M.simSpeed = 1;
    callGame('updateUI');
    return;
  }

  const n = M.simSpeed || 1;
  for (let i = 0; i < n; i++) updateGame(dt);
}

/* ---------------- PER-FRAME UPDATE ---------------- */
let _lastQ = 1;
function update(dt) {
  // Momentum decay toward neutral
  if (Math.abs(M.value) > 0.5) {
    M.value -= Math.sign(M.value) * Math.min(Math.abs(M.value), 2.2 * dt);
  } else M.value = 0;

  // Crowd drifts toward a baseline set by score + momentum
  const diff = G('scoreUser', 0) - G('scoreCPU', 0);
  const baseline = clamp(50 + diff * 1.2 + M.value * 0.25, 8, 100);
  M.crowdEnergy += (baseline - M.crowdEnergy) * Math.min(1, 0.25 * dt);
  renderCrowd();
  renderMeter();

  // Quarter transitions reset short-term run memory slightly
  const q = G('gameQuarter', 1);
  if (q !== _lastQ) {
    _lastQ = q;
    M.value *= 0.5;
    M.runActive = null;
    M.runAlerted = '';
    M._blowout = false;
  }

  checkClutch();
  // Win prob refreshes on scores; also nudge it as clock runs
  if (Math.random() < dt * 0.5) updateWinProb();
}

/** Reset for a new game. Call from begin(). */
function reset() {
  M.value = 0; M.scoreLog = []; M.runAlerted = ''; M.runActive = null;
  M.crowdEnergy = 50; M.clutchOn = false; M.simSpeed = 1; M.simTarget = null;
  M.winProb = 50; M.largestDeficit = 0; M.largestLead = 0;
  M.comebackAlerted = false; M._blowout = false; _lastQ = 1;
  M.els.clutch && M.els.clutch.classList.remove('show');
  renderMeter(); renderCrowd(); renderWP();
}

/* ---------------- INIT ---------------- */
function init() {
  if (document.getElementById('momentumWrap')) return;
  buildDOM();
  renderMeter(); renderCrowd(); renderWP();
  // Auto-reset when a new game begins: wrap begin() if present
  try {
    const origBegin = eval('begin');
    if (typeof origBegin === 'function' && !origBegin._momWrapped) {
      const wrapped = function (...a) { reset(); return origBegin.apply(this, a); };
      wrapped._momWrapped = true;
      eval('begin = wrapped');
    }
  } catch (e) {}
}

if (document.readyState === 'loading') {
  document.addEventListener('DOMContentLoaded', init);
} else {
  init();
}

/* ---------------- PUBLIC API ---------------- */
window.Momentum = {
  onScore, onBigPlay, onTimeout,
  update, tickSim, reset,
  getShootingMult, getCrowdVolumeMult,
  get state() {
    return {
      value: Math.round(M.value),
      crowdEnergy: Math.round(M.crowdEnergy),
      clutch: M.clutchOn,
      winProb: M.winProb,
      run: M.runActive,
      simSpeed: M.simSpeed,
    };
  },
  // Expose internals for debugging / tuning
  _m: M,
};

})();

    tex.colorSpace = T.SRGBColorSpace;
    const sp = new T.Sprite(new T.SpriteMaterial({ map: tex, transparent: true, depthWrite: false }));
    sp.scale.set(0.85, 0.85, 1);
    return sp;
  }

  function makeSnowSprite() {
    const c = document.createElement('canvas');
    c.width = c.height = 128;
    const g = c.getContext('2d');
    g.strokeStyle = '#bfe9ff'; g.lineWidth = 7; g.lineCap = 'round';
    for (let i = 0; i < 3; i++) {
      const a = (i * Math.PI) / 3;
      g.beginPath();
      g.moveTo(64 - 44 * Math.cos(a), 64 - 44 * Math.sin(a));
      g.lineTo(64 + 44 * Math.cos(a), 64 + 44 * Math.sin(a));
      g.stroke();
    }
    g.fillStyle = '#eaf7ff';
    g.beginPath(); g.arc(64, 64, 12, 0, Math.PI * 2); g.fill();
    const tex = new T.CanvasTexture(c);
    tex.colorSpace = T.SRGBColorSpace;
    const sp = new T.Sprite(new T.SpriteMaterial({ map: tex, transparent: true, depthWrite: false, opacity: 0.95 }));
    sp.scale.set(0.7, 0.7, 1);
    return sp;
  }

  function arrowLine(from, to, color, y = 0.06) {
    const grp = new T.Group();
    const mat = new T.LineBasicMaterial({ color, transparent: true, opacity: 0.95 });
    const pts = [new T.Vector3(from[0], y, from[1]), new T.Vector3(to[0], y, to[1])];
    grp.add(new T.Line(new T.BufferGeometry().setFromPoints(pts), mat));
    // arrowhead
    const dir = new T.Vector3(to[0] - from[0], 0, to[1] - from[1]);
    const len = dir.length() || 1; dir.normalize();
    const perp = new T.Vector3(-dir.z, 0, dir.x);
    const tip = new T.Vector3(to[0], y, to[1]);
    const hl = 0.55, hw = 0.32;
    const p1 = tip.clone().addScaledVector(dir, -hl).addScaledVector(perp, hw);
    const p2 = tip.clone().addScaledVector(dir, -hl).addScaledVector(perp, -hw);
    const tri = new T.BufferGeometry().setFromPoints([tip, p1, p2]);
    grp.add(new T.Mesh(tri, new T.MeshBasicMaterial({ color, transparent: true, opacity: 0.95, side: T.DoubleSide })));
    return grp;
  }

  function ringMarker(color, radius = 0.42) {
    const m = new T.Mesh(
      new T.RingGeometry(radius - 0.09, radius, 32),
      new T.MeshBasicMaterial({ color, transparent: true, opacity: 0.95, side: T.DoubleSide })
    );
    m.rotation.x = -Math.PI / 2;
    m.position.y = 0.05;
    return m;
  }

  function showPlayArt(playId, duration = ART_LIFE) {
    if (!T || !scene) return;
    const play = PLAYS[playId] || PLAYS.motion;
    clearArt();
    lastPlayShown = playId;
    artGroup = new T.Group();
    const { playerTeam, cpuTeam, ballTeam } = getState();
    const off = ballTeam === 'user' ? playerTeam : cpuTeam;
    const def = ballTeam === 'user' ? cpuTeam : playerTeam;

    // O markers on offense at their play spots, X markers on defense
    off.forEach((p, i) => {
      const s = play.spots[i % play.spots.length];
      const o = ringMarker(COLORS.offense);
      o.position.set(s[0], 0.05, s[1]);
      artGroup.add(o);
      const label = makeTextSprite(String(i + 1), { size: 52, color: '#d8ff48' });
      label.position.set(s[0], 0.75, s[1]);
      artGroup.add(label);
    });
    def.forEach((p) => {
      const x = makeTextSprite('✕', { size: 72, color: '#ff5a5a' });
      x.position.set(p.position.x, 0.75, p.position.z);
      artGroup.add(x);
    });

    // movement cuts
    (play.cuts || []).forEach(([a, b]) => artGroup.add(arrowLine(a, b, COLORS.arrow)));
    // ball path (dashed feel via short segments)
    const bp = play.ballPath || [];
    for (let i = 0; i < bp.length - 1; i++) {
      artGroup.add(arrowLine(bp[i], bp[i + 1], COLORS.ball));

      sc.position.set(play.screen[0], 0.9, play.screen[1]);

    title.scale.set(4.4, 1.1, 1);
    title.position.set(0, 3.4, 0);

    artTimer = duration;

  function recordShot(x, z, made, team, handler) {
    const s = { x: clamp(x, -14, 14), z: clamp(z, -7.5, 7.5), made: !!made, team, handler };
    shots.push(s);
    if (team === 'user' || team === 'cpu') {
      const arr = playerShots[team][handler] || (playerShots[team][handler] = []);
      arr.push(made ? 1 : 0);
      if (arr.length > 12) arr.shift();
    }
    if (shotChartOn) drawShotChart();
    refreshHotCold();
  }

  function drawShotChart() {
    if (!shotCtx) return;
    const ctx = shotCtx, w = shotCanvas.width, h = shotCanvas.height, pad = 10;
    ctx.clearRect(0, 0, w, h);
    drawMiniCourt(ctx, w, h, pad);
    for (const s of shots) {
      const [px, py] = courtToMini(s.x, s.z, w, h, pad);
      ctx.beginPath();
      ctx.arc(px, py, 3.4, 0, Math.PI * 2);
      if (s.team === 'user') ctx.fillStyle = s.made ? '#7CFC00' : '#ff5a5a';
      else ctx.fillStyle = s.made ? '#7ab8ff' : '#c26a6a';
      ctx.fill();
      if (!s.made) {
        ctx.strokeStyle = 'rgba(0,0,0,.6)'; ctx.lineWidth = 1; ctx.stroke();
      }
    }
  }

  // ------------------------------------------------- 3. matchup display
  function setMatchupsVisible(v) {
    matchupsOn = !!v;

    if (!v) return;
    matchupGroup = new T.Group();
    const { playerTeam, cpuTeam, ballTeam } = getState();
    const def = ballTeam === 'user' ? cpuTeam : playerTeam;
    const off = ballTeam === 'user' ? playerTeam : cpuTeam;
    // nearest-man assignments
    def.forEach((d) => {
      let best = null, bd = 1e9;
      off.forEach((o) => {
        const dist = d.position.distanceTo(o.position);
        if (dist < bd) { bd = dist; best = o; }
      });
      if (!best) return;
      const mat = new T.LineBasicMaterial({ color: COLORS.defense, transparent: true, opacity: 0.4 });
      const geo = new T.BufferGeometry().setFromPoints([
        new T.Vector3(d.position.x, 1.4, d.position.z),
        new T.Vector3(best.position.x, 1.4, best.position.z),
      ]);
      matchupGroup.add(new T.Line(geo, mat));
    });
    scene.add(matchupGroup);
  }

  // ------------------------------------------------- 4. hot / cold
  function heatOf(team, i) {
    const arr = (playerShots[team] && playerShots[team][i]) || [];
    const w = arr.slice(-HOT_WINDOW);
    if (w.length < 3) return 0;
    const makes = w.reduce((a, b) => a + b, 0);
    if (makes >= HOT_MAKES) return 1;
    if (w.length - makes >= COLD_MISSES) return -1;
    return 0;
  }

  function refreshHotCold() {
    if (!T || !scene) return;
    if (hotGroup) { scene.remove(hotGroup); hotGroup = null; }
    hotGroup = new T.Group();
    const { playerTeam, cpuTeam } = getState();
    const place = (p, team, i) => {
      const h = heatOf(team, i);
      if (!h) return;
      const sp = h > 0 ? makeFlameSprite() : makeSnowSprite();
      sp.position.set(p.position.x, 2.85, p.position.z);
      sp.userData.follow = p;
      hotGroup.add(sp);
      const ring = ringMarker(h > 0 ? COLORS.hot : COLORS.cold, 0.5);
      ring.position.set(p.position.x, 0.05, p.position.z);
      ring.userData.follow = p;
      hotGroup.add(ring);
    };
    playerTeam.forEach((p, i) => place(p, 'user', i));
    cpuTeam.forEach((p, i) => place(p, 'cpu', i));
    scene.add(hotGroup);
  }

  function drawPlayDiagram(ctx, playId) {
    const play = PLAYS[playId];
    const w = 200, h = 120, pad = 8;
    ctx.clearRect(0, 0, w, h);
    drawMiniCourt(ctx, w, h, pad);
    // offense O's at spots
    play.spots.forEach(([x, z], i) => {
      const [px, py] = courtToMini(x, z, w, h, pad);
      ctx.beginPath(); ctx.arc(px, py, 7, 0, Math.PI * 2);
      ctx.strokeStyle = '#d8ff48'; ctx.lineWidth = 2; ctx.stroke();
      ctx.fillStyle = '#d8ff48'; ctx.font = '700 8px Arial';
      ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
      ctx.fillText(String(i + 1), px, py);
    });
    // cuts
    ctx.strokeStyle = '#ffffff'; ctx.lineWidth = 1.6;
    (play.cuts || []).forEach(([a, b]) => {
      const [ax, ay] = courtToMini(a[0], a[1], w, h, pad);
      const [bx, by] = courtToMini(b[0], b[1], w, h, pad);
      ctx.beginPath(); ctx.moveTo(ax, ay); ctx.lineTo(bx, by); ctx.stroke();
      const ang = Math.atan2(by - ay, bx - ax);
      ctx.beginPath();
      ctx.moveTo(bx, by);
      ctx.lineTo(bx - 7 * Math.cos(ang - 0.45), by - 7 * Math.sin(ang - 0.45));
      ctx.moveTo(bx, by);
      ctx.lineTo(bx - 7 * Math.cos(ang + 0.45), by - 7 * Math.sin(ang + 0.45));
      ctx.stroke();
    });
    // ball path in orange
    ctx.strokeStyle = '#ffb020'; ctx.lineWidth = 2; ctx.setLineDash([4, 3]);
    ctx.beginPath();
    (play.ballPath || []).forEach(([x, z], i) => {
      const [px, py] = courtToMini(x, z, w, h, pad);
      i ? ctx.lineTo(px, py) : ctx.moveTo(px, py);
    });
    ctx.stroke(); ctx.setLineDash([]);
    // screen marker
    if (play.screen) {
      const [px, py] = courtToMini(play.screen[0], play.screen[1], w, h, pad);
      ctx.fillStyle = '#ffffff'; ctx.font = '900 11px Arial';
      ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
      ctx.fillText('S', px, py - 12);
    }
  }

  // --------------------------------------- 6. defensive scheme visualization
  function refreshZoneOverlay() {
    if (!T || !scene) return;
    if (zoneGroup) { scene.remove(zoneGroup); zoneGroup = null; }
    const { defense, ballTeam } = getState();
    // 'paint' (protect the paint) and 'switch' get zone shading; man/pressure use matchup lines
    if (defense !== 'paint' && defense !== 'switch') return;
    zoneGroup = new T.Group();
    const mat = new T.MeshBasicMaterial({ color: COLORS.zone, transparent: true, opacity: 0.13, side: T.DoubleSide, depthWrite: false });
    // zone boxes on the defensive half (defense defends -x basket when ballTeam==='user')
    const side = ballTeam === 'user' ? -1 : 1;
    const zones = defense === 'paint'
      ? [[-9, 0, 10, 12], [-3.5, -4, 5, 6], [-3.5, 4, 5, 6]]   // packed 2-3-ish
      : [[-11, -5, 6, 5], [-11, 5, 6, 5], [-4, 0, 8, 14]];      // extended switch shell
    zones.forEach(([cx, cz, w, d]) => {
      const m = new T.Mesh(new T.PlaneGeometry(w, d), mat);
      m.rotation.x = -Math.PI / 2;
      m.position.set(side * Math.abs(cx), 0.04, cz);
      zoneGroup.add(m);
      const edge = new T.LineSegments(
        new T.EdgesGeometry(new T.PlaneGeometry(w, d)),
        new T.LineBasicMaterial({ color: COLORS.zone, transparent: true, opacity: 0.5 })
      );
      edge.rotation.x = -Math.PI / 2;
      edge.position.copy(m.position);
      zoneGroup.add(edge);
    });
    scene.add(zoneGroup);
  }

  // ---------------------------------------------------------------- update
  let lastDefSeen = null;
  function update(dt) {
    // fade + retire play art
    if (artGroup && artTimer > 0) {
      artTimer -= dt;
      const k = clamp(artTimer / 0.6, 0, 1); // fade last 0.6s
      artGroup.traverse((o) => {
        if (o.material && o.material.transparent) o.material.opacity = Math.min(o.material.opacity, k);
      });
      if (artTimer <= 0) clearArt();
    }
    // keep hot/cold markers glued to players
    if (hotGroup) {
      hotGroup.children.forEach((s) => {
        if (s.userData.follow) s.position.set(s.userData.follow.position.x, s.position.y, s.userData.follow.position.z);
      });
      // gentle bob on flames
      const t = performance.now() / 1000;
      hotGroup.children.forEach((s, i) => {
        if (s.isSprite) s.position.y += Math.sin(t * 3 + i) * 0.0015;
      });
    }
    // refresh matchup lines positions (cheap: rebuild every ~0.5s)
    update._mt = (update._mt || 0) + dt;
    if (matchupsOn && update._mt > 0.5) {
      update._mt = 0;
      setMatchupsVisible(true);
    }
    // watch for defensive scheme changes
    const { defense } = getState();
    if (defense !== lastDefSeen) {
      lastDefSeen = defense;
      refreshZoneOverlay();
      // man schemes pair naturally with matchup lines
      if ((defense === 'man' || defense === 'pressure') && !matchupsOn) {
        // leave toggle to the coach; nothing automatic
      }
    }
  }

  // ------------------------------------------------------------------ api
  function init(options = {}) {
    opts = options || {};
    T = opts.THREE || (typeof window !== 'undefined' && window.THREE) || null;
    scene = opts.scene || null;
    if (!T) throw new Error('[PlayVisuals] THREE namespace required: PlayVisuals.init({ THREE, scene })');
    if (!scene) throw new Error('[PlayVisuals] scene required: PlayVisuals.init({ THREE, scene })');
    ensureUI();
    lastDefSeen = getState().defense;
    refreshZoneOverlay();
    return api;
  }

  const api = {
    init,
    update,
    showPlayArt,
    clearArt,
    recordShot,
    setShotChartVisible,
    setMatchupsVisible,
    openPlayMenu,
    closePlayMenu,
    refreshHotCold,
    refreshZoneOverlay,
    heatOf,
    PLAYS: () => JSON.parse(JSON.stringify(PLAYS)),
    getShots: () => shots.slice(),
    clearShots: () => { shots.length = 0; for (const t of ['user', 'cpu']) playerShots[t] = [[],[],[],[],[]]; if (shotCtx) drawShotChart(); refreshHotCold(); },
  };
  return api;
})();

function blankPlayer(name, number, pos) {
  return {
    name: name, number: number || 0, pos: pos || '',
    min: 0,                       // seconds on floor
    pts: 0, fgm: 0, fga: 0, tpm: 0, tpa: 0, ftm: 0, fta: 0,
    oreb: 0, dreb: 0, ast: 0, stl: 0, blk: 0, tov: 0, pf: 0,
    plusMinus: 0,
    zones: { paint: { m: 0, a: 0 }, mid: { m: 0, a: 0 }, three: { m: 0, a: 0 } }
  };
}

function blankTeam(cfg) {
  var players = {};
  (cfg.roster || []).forEach(function (r) {
    players[r.name] = blankPlayer(r.name, r.number, r.pos);
  });
  return {
    abbr: cfg.abbr, name: cfg.name, color: cfg.color || '#f3b544',
    players: players, order: (cfg.roster || []).map(function (r) { return r.name; }),
    lineup: [], lineupStart: 0, lineups: {},   // key -> {names, min, plus, minus}
    fastBreakPts: 0, paintPts: 0,
    run: 0, lastScorer: null,                  // scoring-run tracking
    lastPts: 0
  };
}

function now() { return Date.now() / 1000; }

function ensurePlayer(team, name) {
  if (!team.players[name]) {
    team.players[name] = blankPlayer(name, 0, '');
    team.order.push(name);
  }
  return team.players[name];
}

/* ============================================================
   EVENT INGESTION + DERIVED STATS
   team: 'home' | 'away'
   ============================================================ */

function T(side) { return side === 'home' ? S.home : S.away; }
function O(side) { return side === 'home' ? S.away : S.home; }

/* --- lineups & plus/minus --------------------------------------------
   The game swaps roster identity onto fixed models (coachSubstitute),
   so lineups are tracked by player NAME, set explicitly via setLineup()
   or incrementally via substitution(). Floor time accrues per player.  */
function stampFloorTime() {
  var t = now(), dt;
  ['home', 'away'].forEach(function (side) {
    var tm = T(side);
    if (!tm.lineup.length) return;
    dt = t - tm.lineupStart;
    tm.lineup.forEach(function (nm) { ensurePlayer(tm, nm).min += dt; });
    var k = lineupKey(tm.lineup);
    if (tm.lineups[k]) tm.lineups[k].min += dt;
    tm.lineupStart = t;
  });
}

function applyScore(teamSide, pts) {
  stampFloorTime();
  var tm = T(teamSide), opp = O(teamSide);
  tm.lastPts += pts;
  // run tracking: unanswered points by one team
  if (tm.lastScorer === teamSide || tm.lastScorer === null && opp.lastScorer === null) {
    tm.run += pts;
  } else {
    tm.run = pts;
  }
  tm.lastScorer = teamSide;
  opp.lastScorer = null; opp.run = 0;
  // lineup +/-
  [tm, opp].forEach(function (tx, i) {
    var diff = (i === 0 ? pts : -pts);
    tx.lineup.forEach(function (nm) { ensurePlayer(tx, nm).plusMinus += diff; });
    var k = lineupKey(tx.lineup);
    if (tx.lineups[k]) { tx.lineups[k].plus += (i === 0 ? pts : 0); tx.lineups[k].minus += (i === 0 ? 0 : pts); }
  });
  checkRunAlert(tm);
}

function zoneOf(points, kind) {
  if (points === 3) return 'three';
  if (kind === 'LAYUP' || kind === 'DUNK') return 'paint';
  return 'mid';
}

function checkMilestones(tm, p) {
  var q = [];
  if (p.pts === 20 || p.pts === 30 || p.pts === 40) q.push({ t: p.name.toUpperCase() + ': ' + p.pts + ' PTS', s: fgLine(p) + ' FG' });
  if (p.pts >= 10 && p.reb() >= 10 && !p._dd) { p._dd = true; q.push({ t: 'DOUBLE-DOUBLE', s: p.name + ' · ' + p.pts + ' PTS, ' + p.reb() + ' REB' }); }
  if (p.ast === 10 && !p._a10) { p._a10 = true; q.push({ t: '10 ASSISTS', s: p.name + ' running the show' }); }
  if (p.stl === 5 && !p._s5) { p._s5 = true; q.push({ t: '5 STEALS', s: p.name + ' wreaking havoc' }); }
  if (p.blk === 5 && !p._b5) { p._b5 = true; q.push({ t: '5 BLOCKS', s: p.name + ' protecting the rim' }); }
  q.forEach(function (a) { broadcast(a.t, a.s, tm.color, 'MILESTONE'); });
}

function checkRunAlert(tm) {
  if (tm.run >= 8 && !tm._runFired) {
    tm._runFired = true;
    broadcast(tm.abbr + ': ' + tm.run + '-0 RUN', 'Momentum swinging ' + (tm === S.home ? 'home' : 'away') + ' · timeout territory', tm.color, 'SCORING RUN');
  }
  if (tm.run < 8) tm._runFired = false;
}

function fgLine(p) { return p.fgm + '-' + p.fga; }

/* --- public event hooks ---------------------------------------------- */
var API = {};

API.score = function (ev) {
  // ev: {team, player, points, kind, fastBreak?, assistedBy?}
  if (!S) return;
  var tm = T(ev.team), p = ensurePlayer(tm, ev.player);
  var z = zoneOf(ev.points, ev.kind);
  p.pts += ev.points; p.fgm++; p.fga++;
  p.zones[z].m++; p.zones[z].a++;
  if (ev.points === 3) { p.tpm++; p.tpa++; }
  if (z === 'paint') tm.paintPts += ev.points;
  if (ev.fastBreak) tm.fastBreakPts += ev.points;
  if (ev.assistedBy) ensurePlayer(tm, ev.assistedBy).ast++;
  applyScore(ev.team, ev.points);
  checkMilestones(tm, p);
  API.render();
};

API.miss = function (ev) {
  // ev: {team, player, points, kind}
  if (!S) return;
  var tm = T(ev.team), p = ensurePlayer(tm, ev.player);
  var z = zoneOf(ev.points, ev.kind);
  p.fga++; p.zones[z].a++;
  if (ev.points === 3) p.tpa++;
  API.render();
};

API.freeThrow = function (ev) {
  // ev: {team, player, made, attempts}
  if (!S) return;
  var tm = T(ev.team), p = ensurePlayer(tm, ev.player);
  p.ftm += ev.made; p.fta += ev.attempts;
  if (ev.made > 0) { p.pts += ev.made; applyScore(ev.team, ev.made); }
  API.render();
};

API.rebound = function (ev) {
  // ev: {team, player, offensive}
  if (!S) return;
  var p = ensurePlayer(T(ev.team), ev.player);
  if (ev.offensive) p.oreb++; else p.dreb++;
  checkMilestones(T(ev.team), p);
  API.render();
};

API.assist = function (ev) {
  if (!S) return;
  var p = ensurePlayer(T(ev.team), ev.player);
  p.ast++;
  checkMilestones(T(ev.team), p);
  API.render();
};

API.steal = function (ev) {
  if (!S) return;
  var p = ensurePlayer(T(ev.team), ev.player);
  p.stl++;
  ensurePlayer(O(ev.team), ev.victim || '').tov += ev.victim ? 1 : 0;
  checkMilestones(T(ev.team), p);
  broadcast('STEAL', p.name + ' · ' + p.stl + ' STL', T(ev.team).color, 'HUSTLE');
  API.render();
};

API.block = function (ev) {
  if (!S) return;
  var p = ensurePlayer(T(ev.team), ev.player);
  p.blk++;
  checkMilestones(T(ev.team), p);
  broadcast('REJECTED', p.name + ' · ' + p.blk + ' BLK', T(ev.team).color, 'DEFENSE');
  API.render();
};

API.turnover = function (ev) {
  // ev: {team, player, type}
  if (!S) return;
  ensurePlayer(T(ev.team), ev.player).tov++;
  API.render();
};

API.foul = function (ev) {
  if (!S) return;
  var p = ensurePlayer(T(ev.team), ev.player);
  p.pf++;
  if (p.pf === 5) broadcast('FOUL TROUBLE', p.name + ' · 5 PF', T(ev.team).color, 'FOULS');
  if (p.pf >= 6) broadcast('FOULED OUT', p.name + ' is done for the night', T(ev.team).color, 'FOULS');
  API.render();
};

API.setLineup = function (teamSide, names) {
  if (!S) return;
  stampFloorTime();
  var tm = T(teamSide);
  tm.lineup = names.slice();
  tm.lineupStart = now();
  var k = lineupKey(names);
  if (!tm.lineups[k]) tm.lineups[k] = { names: names.slice(), min: 0, plus: 0, minus: 0 };
  names.forEach(function (nm) { ensurePlayer(tm, nm); });
  API.render();
};

API.substitution = function (ev) {
  // ev: {team, out, in}
  if (!S) return;
  var tm = T(ev.team);
  var nl = tm.lineup.map(function (nm) { return nm === ev.out ? ev.in : nm; });
  API.setLineup(ev.team, nl);
  broadcast('SUBSTITUTION', ev.in + ' in for ' + ev.out, tm.color, 'ROTATION');
};

API.endQuarter = function (q) {
  if (!S) return;
  S.quarter = q + 1;
  API.render();
};

API.reset = function () {
  if (!S) return;
  var cfg = S._cfg;
  initState(cfg);
  API.renderAll();
};

/* --- derived team totals ---------------------------------------------- */
function teamTotals(tm) {
  var t = { pts: 0, fgm: 0, fga: 0, tpm: 0, tpa: 0, ftm: 0, fta: 0,
            oreb: 0, dreb: 0, ast: 0, stl: 0, blk: 0, tov: 0, pf: 0,
            paint: { m: 0, a: 0 }, mid: { m: 0, a: 0 }, three: { m: 0, a: 0 } };
  Object.keys(tm.players).forEach(function (k) {
    var p = tm.players[k];
    t.pts += p.pts; t.fgm += p.fgm; t.fga += p.fga; t.tpm += p.tpm; t.tpa += p.tpa;
    t.ftm += p.ftm; t.fta += p.fta; t.oreb += p.oreb; t.dreb += p.dreb;
    t.ast += p.ast; t.stl += p.stl; t.blk += p.blk; t.tov += p.tov; t.pf += p.pf;
    ['paint', 'mid', 'three'].forEach(function (z) { t[z].m += p.zones[z].m; t[z].a += p.zones[z].a; });
  });
  t.reb = t.oreb + t.dreb;
  return t;
}

function pct(m, a) { return a > 0 ? (100 * m / a) : 0; }
function fmtPct(m, a) { return a > 0 ? (100 * m / a).toFixed(1) : '—'; }
function fmtClock(sec) {
  sec = Math.floor(sec);
  return Math.floor(sec / 60) + ':' + String(sec % 60).padStart(2, '0');
}
// rebound accessor attached lazily (kept off the blank template for JSON safety)
function withHelpers(p) {
  if (!p.reb) p.reb = function () { return this.oreb + this.dreb; };
  return p;
}

function el(tag, cls, html) {
  var d = document.createElement(tag);
  if (cls) d.className = cls;
  if (html != null) d.innerHTML = html;
  return d;
}

function esc(s) { return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;'); }

function pmClass(v) { return v > 0 ? 'cs27-pm-pos' : (v < 0 ? 'cs27-pm-neg' : ''); }
function pmText(v) { return (v > 0 ? '+' : '') + v; }

/* --- box score table for one team --- */
function boxTable(tm) {
  var t = el('table', 'cs27-table num');
  var html = '<tr class="cs27-teamhead"><td colspan="14"><span class="dot" style="background:' +
    tm.color + '"></span>' + esc(tm.abbr) + ' · ' + esc(tm.name) + '</td></tr>';
  html += '<tr><th>PLAYER</th><th>MIN</th><th>PTS</th><th>REB</th><th>AST</th><th>STL</th><th>BLK</th>' +
          '<th>TO</th><th>PF</th><th>FG</th><th>3P</th><th>FT</th><th>FG%</th><th>+/-</th></tr>';
  var rows = tm.order.map(function (nm) { return withHelpers(tm.players[nm]); })
    .filter(Boolean)
    .sort(function (a, b) { return b.pts - a.pts || b.min - a.min; });
  rows.forEach(function (p, i) {
    html += '<tr' + (i % 2 ? '' : ' class="hl"') + '><td><span class="cs27-pos">' + esc(p.pos) +
      '</span><span class="cs27-pname">' + esc(p.name) + '<small>#' + p.number + '</small></span></td>' +
      '<td>' + fmtClock(p.min) + '</td><td><b>' + p.pts + '</b></td>' +
      '<td>' + p.reb() + '</td><td>' + p.ast + '</td><td>' + p.stl + '</td><td>' + p.blk + '</td>' +
      '<td>' + p.tov + '</td><td>' + p.pf + '</td>' +
      '<td>' + p.fgm + '-' + p.fga + '</td><td>' + p.tpm + '-' + p.tpa + '</td><td>' + p.ftm + '-' + p.fta + '</td>' +
      '<td>' + fmtPct(p.fgm, p.fga) + '</td>' +
      '<td class="' + pmClass(p.plusMinus) + '">' + pmText(p.plusMinus) + '</td></tr>';
  });
  var tt = teamTotals(tm);
  html += '<tr class="tot"><td>TEAM</td><td></td><td>' + tt.pts + '</td><td>' + tt.reb + '</td><td>' + tt.ast +
    '</td><td>' + tt.stl + '</td><td>' + tt.blk + '</td><td>' + tt.tov + '</td><td>' + tt.pf + '</td>' +
    '<td>' + tt.fgm + '-' + tt.fga + '</td><td>' + tt.tpm + '-' + tt.tpa + '</td><td>' + tt.ftm + '-' + tt.fta +
    '</td><td>' + fmtPct(tt.fgm, tt.fga) + '</td><td></td></tr>';
  t.innerHTML = html;
  return t;
}

/* --- team comparison bars --- */
function cmpRow(label, hv, av, fmt, higherWins) {
  if (higherWins === undefined) higherWins = true;
  var hWin = higherWins ? hv >= av : hv <= av;
  var aWin = higherWins ? av >= hv : av <= hv;
  var mx = Math.max(Math.abs(hv), Math.abs(av), 1e-6);
  function side(v, color, win, rev) {
    return '<div class="cs27-cside' + (rev ? ' r' : '') + '"><span class="cs27-cval num' +
      (win && hv !== av ? ' cs27-win' : '') + '">' + fmt(v) + '</span>' +
      '<div class="cs27-cbar"><i style="width:' + (100 * Math.abs(v) / mx).toFixed(1) +
      '%;background:' + color + '"></i></div></div>';
  }
  var row = el('div', 'cs27-crow');
  row.innerHTML = side(hv, S.home.color, hWin, false) +
    '<div class="lab">' + label + '</div>' + side(av, S.away.color, aWin, true);
  return row;
}

function comparePanel() {
  var wrap = el('div', 'cs27-cmp cs27');
  var H = teamTotals(S.home), A = teamTotals(S.away);
  var f1 = function (v) { return v.toFixed(1); }, fi = function (v) { return String(Math.round(v)); };
  wrap.appendChild(cmpRow('POINTS', H.pts, A.pts, fi));
  wrap.appendChild(cmpRow('FIELD GOAL %', pct(H.fgm, H.fga), pct(A.fgm, A.fga), f1));
  wrap.appendChild(cmpRow('THREE POINT %', pct(H.tpm, H.tpa), pct(A.tpm, A.tpa), f1));
  wrap.appendChild(cmpRow('FREE THROW %', pct(H.ftm, H.fta), pct(A.ftm, A.fta), f1));
  wrap.appendChild(cmpRow('REBOUNDS', H.reb, A.reb, fi));
  wrap.appendChild(cmpRow('ASSISTS', H.ast, A.ast, fi));
  wrap.appendChild(cmpRow('STEALS', H.stl, A.stl, fi));
  wrap.appendChild(cmpRow('BLOCKS', H.blk, A.blk, fi));
  wrap.appendChild(cmpRow('TURNOVERS', H.tov, A.tov, fi, false));
  wrap.appendChild(cmpRow('FOULS', H.pf, A.pf, fi, false));
  wrap.appendChild(cmpRow('PTS IN PAINT', S.home.paintPts, S.away.paintPts, fi));
  wrap.appendChild(cmpRow('FAST BREAK PTS', S.home.fastBreakPts, S.away.fastBreakPts, fi));
  var note = el('div', 'cs27-note');
  note.textContent = 'GOLD = LEADING · TURNOVERS & FOULS: LOWER WINS';
  wrap.appendChild(note);
  return wrap;
}

/* --- shot zone court (SVG half-court, paint / mid / three) --- */
function zoneCourt(tm) {
  // 300x280 half-court: paint rect, FT circle, three arc
  var z = { paint: teamTotals(tm).paint, mid: teamTotals(tm).mid, three: teamTotals(tm).three };
  function heat(m, a) {
    if (!a) return '#101820';
    var e = m / a, op = 0.25 + 0.65 * Math.min(1, a / 8);
    var c = e >= 0.5 ? '62,220,120' : (e >= 0.38 ? '243,181,68' : '255,110,110');
    return 'rgba(' + c + ',' + op.toFixed(2) + ')';
  }
  var svg =
    '<svg viewBox="0 0 300 280" width="230" style="background:#0a0f15;border:1px solid #1e2836;border-radius:4px">' +
    '<rect x="8" y="8" width="284" height="264" fill="none" stroke="#3a4a5c" stroke-width="2"/>' +

/* --- lineups --- */
function lineupsPanel() {
  var wrap = el('div', 'cs27');
  ['home', 'away'].forEach(function (side) {
    var tm = T(side);
    var t = el('table', 'cs27-table num');
    var html = '<tr class="cs27-teamhead"><td colspan="5"><span class="dot" style="background:' +
      tm.color + '"></span>' + esc(tm.abbr) + ' · 5-MAN UNITS</td></tr>' +
      '<tr><th>LINEUP</th><th>MIN</th><th>+</th><th>−</th><th>NET</th></tr>';
    var lus = Object.keys(tm.lineups).map(function (k) { return tm.lineups[k]; })
      .sort(function (a, b) { return b.min - a.min; });
    if (!lus.length) html += '<tr><td colspan="5" style="text-align:center;color:#5a6a7c">No lineup data yet — set via CourtsideStats.setLineup()</td></tr>';
    lus.forEach(function (lu, i) {
      var net = lu.plus - lu.minus;
      var short = lu.names.map(function (n) { return esc(n.split(' ').pop()); }).join(' · ');
      html += '<tr' + (i % 2 ? '' : ' class="hl"') + '><td class="cs27-lu" style="text-align:left">' + short +
        '</td><td>' + fmtClock(lu.min) + '</td><td>' + lu.plus + '</td><td>' + lu.minus +
        '</td><td class="' + pmClass(net) + '"><b>' + pmText(net) + '</b></td></tr>';
    });
    var cur = el('div', 'cs27-note');
    cur.textContent = 'ON FLOOR NOW: ' + (tm.lineup.length ? tm.lineup.map(function (n) { return n.split(' ').pop(); }).join(' · ').toUpperCase() : '—');
    t.innerHTML = html;
    wrap.appendChild(t);
    wrap.appendChild(cur);
  });
  return wrap;
}

var dashEls = {};
function buildDashboard() {
  injectCSS();
  var root = el('div', 'cs27');
  root.id = 'cs27-dash';
  var card = el('div', 'cs27-card');
  var head = el('div', 'cs27-head');
  head.innerHTML = '<h2>COACH\'S <b>DASHBOARD</b></h2>' +
    '<div class="cs27-score num"><span id="cs27-dh" style="color:' + S.home.color + '"></span> <span style="color:#5a6a7c">·</span> <span id="cs27-da" style="color:' + S.away.color + '"></span></div>';
  var close = el('button', 'cs27-close', 'CLOSE&nbsp;&nbsp;[S]');
  close.onclick = API.toggleDashboard;
  head.appendChild(close);
  var tabs = el('div', 'cs27-tabs');
  var body = el('div', 'cs27-body');
  var panes = {};
  [['box', 'BOX SCORE'], ['teams', 'TEAM STATS'], ['zones', 'SHOT ZONES'], ['lineups', 'LINEUPS']].forEach(function (td) {
    var b = el('button', 'cs27-tab' + (td[0] === 'box' ? ' on' : ''), td[1]);
    b.onclick = function () { API.showPanel(td[0]); };
    tabs.appendChild(b);
    panes[td[0]] = b;
  });
  card.appendChild(head); card.appendChild(tabs); card.appendChild(body);
  root.appendChild(card);
  document.body.appendChild(root);
  dashEls = { root: root, body: body, panes: panes, panel: 'box' };
  root.addEventListener('click', function (e) { if (e.target === root) API.toggleDashboard(); });
}
