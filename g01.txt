    for (const [k,s] of this.map) if (k.startsWith(team+':')) { pts+=s.pts; fg3+=s.fg3; }

const recorder = {
  frames: [],       // [{t, ball:[x,y,z], actors:[{team,idx,x,y,z,ry}]}]
  acc: 0,
  interval: 1/30,
  maxAge: 6.0,
  record(dt){
    this.acc += dt;
    if (this.acc < this.interval) return;
    this.acc = 0;
    const now = performance.now()/1000;
    const snap = { t: now, ball: null, actors: [] };
    try {
      const bg = G('ballGroup', null);
      if (bg && bg.position) snap.ball = [bg.position.x, bg.position.y, bg.position.z];
      for (const team of ['user','cpu']) {
        const arr = team==='user' ? G('playerTeam',[]) : G('cpuTeam',[]);
        for (let i=0;i<arr.length;i++){
          const p = arr[i]; if(!p||!p.position) continue;
          snap.actors.push({team, idx:i, x:p.position.x, y:p.position.y, z:p.position.z, ry:p.rotation?p.rotation.y:0});
        }
      }
    } catch(e){}
    this.frames.push(snap);
    const cutoff = now - this.maxAge;
    while (this.frames.length && this.frames[0].t < cutoff) this.frames.shift();
  },
  /** snapshot closest to (now - secondsAgo) */
  at(secondsAgo){
    const target = performance.now()/1000 - secondsAgo;
    let best = this.frames[0] || null;
    for (const f of this.frames){
      if (f.t <= target) best = f; else break;
    }
    return best;
  },
  clear(){ this.frames.length = 0; },
};

    const p = arr[idx]; if (p && p.position) return V3(p.position.x, p.position.y, p.position.z);
  } catch(e){}
  return V3(0,1,0);
}
function teamInfo(team){
  try {
    const ct = G('chosenTeam', null);
    if (ct) { const t = ct(team); return { name:t.name||team, abbr:t.abbr||'', color:t.color||'#d8ff48' }; }
  } catch(e){}
  return { name: team, abbr:'', color:'#d8ff48' };
}
function rosterName(team, idx){
  try {
    const rf = G('rosterFor', null);
    if (rf) { const r = rf(team, idx); return { name:r.name||('Player '+(idx+1)), number:r.number||0, pos:r.pos||'' }; }
  } catch(e){}
  return { name:'Player '+(idx+1), number:0, pos:'' };
}

/* Shot definitions: each returns {pos, look, blend} — blend = lerp speed */
const Shots = {
  /** Low baseline camera behind the hoop — for dunks, finishes, replays */
  baselineLow(focus, attacking){
    const hx = HOOP_X * (attacking || 1);
    return {
      pos: V3(hx + 3.4*(attacking||1), 2.0, 7.5),
      look: V3(hx - 2.0*(attacking||1), 2.6, 0),
      blend: 5.2, name:'BASELINE_LOW',
    };
  },
  /** Opposite baseline, slightly elevated — alternate replay angle */
  baselineHigh(focus, attacking){
    const hx = HOOP_X * (attacking || 1);
    return {
      pos: V3(hx + 5.5*(attacking||1), 5.2, -8.5),
      look: V3(hx - 3.0*(attacking||1), 2.2, 0),
      blend: 4.6, name:'BASELINE_HIGH',
    };
  },
  /** High tactical overhead — timeouts, halftime, coach film room */
  overhead(){
    return { pos: V3(0, 26, 6.5), look: V3(0, 0, -0.5), blend: 2.2, name:'OVERHEAD' };
  },
  /** Sideline iso on a single player — clutch moments, star intro */
  isoStar(team, idx){
    const p = getActorPos(team, idx);
    const side = p.z >= 0 ? 1 : -1;
    return {
      pos: V3(p.x - 4.2, 3.4, p.z + side*7.8),
      look: V3(p.x, 1.7, p.z),
      blend: 3.4, name:'ISO_STAR',
    };
  },
  /** Coach huddle — sideline, low, toward bench (behind user basket) */
  huddle(){
    return { pos: V3(-16.5, 3.2, 10.5), look: V3(-9.5, 1.4, 3.5), blend: 2.6, name:'HUDDLE' };
  },
  /** Free-throw side angle */
  freeThrow(side){
    const hx = HOOP_X * side;
    return {
      pos: V3(hx - side*7.0, 3.6, 9.0),
      look: V3(hx - side*1.5, 2.2, 0),
      blend: 3.8, name:'FT_SIDE',
    };
  },
  /** Victory wide — pulls back to show the whole arena */
  celebration(){
    return { pos: V3(0, 13.5, 26), look: V3(0, 1.5, 0), blend: 1.8, name:'CELEBRATION' };
  },
  /** Replay arc — computed per-frame during replay (see director) */
  replayArc(center, radius, angle){
    return {
      pos: V3(center.x + Math.cos(angle)*radius, 4.6, center.z + Math.sin(angle)*radius),
      look: V3(center.x, 2.2, center.z),
      blend: 99, name:'REPLAY_ARC', // blend unused; position set directly
    };
  },
};

/* ------------------------------------------------------------------ */
/* 5. Director state machine                                            */
/* ------------------------------------------------------------------ */
const director = {
  active: false,          // true when director owns the camera
  shot: null,            // current shot {pos, look, blend, name}
  shotT: 0,              // time in current shot
  queue: [],             // upcoming beats: [{at, fn}]
  holdUntil: 0,
  mode: 'live',          // live | replay | timeout | halftime | break | buzzer
  replay: null,          // active replay state
  timeout: null,
  halftime: null,
  buzzer: null,

  clock: 0, // director-local time (advances with update(dt), pause-safe)
  now(){ return this.clock; },

  /** Take over the camera with a shot object */
  cut(shot, mode){
    this.shot = shot;
    this.shotT = 0;
    this.active = true;
    if (mode) this.mode = mode;
    log('CUT ->', shot.name, '| mode:', this.mode);
  },
  /** Release back to the game's own camera */
  release(){
    this.active = false;
    this.shot = null;
    this.mode = 'live';
    this.queue.length = 0;
    log('release -> game camera');
  },
  /** Schedule a beat relative to now */
  after(sec, fn){ this.queue.push({ at: this.now()+sec, fn }); },

  update(dt){
    const sdt = Math.max(dt, 0);
    this.clock += sdt;
    this.shotT += sdt;
    // run scheduled beats
    const n = this.now();
    for (let i=this.queue.length-1;i>=0;i--){
      if (this.queue[i].at <= n){ const b=this.queue.splice(i,1)[0]; try{ b.fn(); }catch(e){ log('beat err', e.message); } }
    }
    if (this.mode === 'replay' && this.replay) this.updateReplay(dt);
  },

  hasOverride(){ return this.active && !!this.shot; },

  applyCamera(dt){
    if (!this.hasOverride()) return;
    const cam = G('camera', null);
    if (!cam) return;
    const s = this.shot;
    if (s.name === 'REPLAY_ARC') {
      // hard-set during replay sweep (positions computed per frame)
      cam.position.copy(s.pos);
      const cl = G('camLook', null);
      if (cl) { cl.copy(s.look); cam.lookAt(cl); } else cam.lookAt(s.look);
      return;
    }
    const kp = 1 - Math.exp(-(s.blend||3.2)*Math.max(dt,.001));
    cam.position.lerp(s.pos, kp);
    try {
      const cl = G('camLook', null);
      if (cl){ cl.lerp(s.look, 1-Math.exp(-4.6*Math.max(dt,.001))); cam.lookAt(cl); }
      else cam.lookAt(s.look);
    } catch(e){ cam.lookAt(s.look); }
  },

  /* ---------------- Replay engine ---------------- */
  startReplay(play){
    // play: {label, team, idx, points, kind, loc:V3, dur}
    if (this.mode === 'replay') return;
    if (!recorder.frames.length || recorder.frames[recorder.frames.length-1].actors.length === 0){
      log('replay: no history, dramatic hold instead');
      this.dramaticHold(play);
      return;
    }
    // stash live transforms
    const live = [];
    try {
      const bg = G('ballGroup', null);
      if (bg) live.push({o:bg.position, v:[bg.position.x,bg.position.y,bg.position.z]});
      for (const team of ['user','cpu']){
        const arr = team==='user'?G('playerTeam',[]):G('cpuTeam',[]);
        arr.forEach((p)=>{ if(p&&p.position) live.push({o:p.position, v:[p.position.x,p.position.y,p.position.z], r:p.rotation, ry:p.rotation?p.rotation.y:0}); });
      }
    } catch(e){}
    const totalFrames = recorder.frames.length - 1;
    // slow-mo: show the most recent ACTION_SPAN seconds stretched over dur
    const ACTION_SPAN = 2.6;
    const spanFrames = clamp(Math.floor(ACTION_SPAN / recorder.interval), 8, totalFrames);
    this.replay = {
      play, live,
      t: 0,
      dur: play.dur || 4.2,
      startFrame: Math.max(0, totalFrames - spanFrames),
      endFrame: totalFrames,
      center: play.loc ? play.loc.clone() : V3(0,2,0),
      angle0: Math.atan2(8, 6),
      radius: 10.5,
    };
    this.mode = 'replay';
    gfx.replayBug(true);
    const rn = rosterName(play.team, play.idx);
    gfx.lowerThird(rn.name, rn.number, `${play.label}  ·  ${stats.line(play.team, play.idx)}`);
    sting('replay');
    log('REPLAY start:', play.label, 'frames', this.replay.startFrame, '->', this.replay.endFrame);
  },

  updateReplay(dt){
    const r = this.replay;
    if (!r) return;
    r.t += dt;
    const span = r.endFrame - r.startFrame;
    const fi = clamp(Math.floor(r.startFrame + (r.t / r.dur) * span), 0, recorder.frames.length - 1);
    const snap = recorder.frames[fi];
    if (snap){
      try {
        const bg = G('ballGroup', null);
        if (bg && snap.ball) bg.position.set(snap.ball[0], snap.ball[1], snap.ball[2]);
        const byTeam = { user: G('playerTeam',[]), cpu: G('cpuTeam',[]) };
        for (const a of snap.actors){
          const p = byTeam[a.team] && byTeam[a.team][a.idx];
          if (p && p.position){ p.position.set(a.x, a.y, a.z); if (p.rotation) p.rotation.y = a.ry; }
        }
      } catch(e){}
    }
    // cinematic arc around the play
    const ang = r.angle0 + (r.t / r.dur) * Math.PI * 0.9;
    this.shot = Shots.replayArc(r.center, r.radius, ang);
    this.active = true;
    if (r.t >= r.dur) this.endReplay();
  },

  endReplay(){
    const r = this.replay;
    if (r){
      try {
        for (const s of r.live){
          s.o.set(s.v[0], s.v[1], s.v[2]);
          if (s.r) s.r.y = s.ry;
        }
      } catch(e){}
    }
    this.replay = null;
    gfx.replayBug(false);
    gfx.hideLowerThird();
    // hand back to live — cut to a fresh sideline angle for energy
    this.mode = 'live';
    this.release();
    log('REPLAY end -> live');
  },

  /** Fallback when no history: dramatic slow push on the moment */
  dramaticHold(play){
    const p = getActorPos(play.team, play.idx);
    this.cut(Shots.isoStar(play.team, play.idx), 'live');
    gfx.replayBug(true);
    const rn = rosterName(play.team, play.idx);
    gfx.lowerThird(rn.name, rn.number, `${play.label}  ·  ${stats.line(play.team, play.idx)}`);
    this.after(3.4, ()=>{ gfx.replayBug(false); this.release(); });
  },

  /* ---------------- Timeout sequence ---------------- */
  startTimeout(side){
    if (this.mode !== 'live') return;
    const ti = teamInfo(side);
    this.mode = 'timeout';
    this.timeout = { side, t:0 };
    // 1. cut to huddle, telestrator tag on
    this.cut(Shots.huddle(), 'timeout');
    gfx.telestrator(true);
    gfx.lowerThird(ti.name + ' TIMEOUT', ti.abbr || 'TO', 'COACH DRAWING UP THE NEXT POSSESSION');
    sting('timeout');
    // 2. after 3s: overhead all-22 + stat blast
    this.after(3.0, ()=>{
      this.cut(Shots.overhead(), 'timeout');
      const tot = stats.teamTotals(side);
      const opp = side==='user'?'cpu':'user';
      const su = G('scoreUser',0), sc = G('scoreCPU',0);
      gfx.blast('TIMEOUT · ' + ti.name,
        `${ti.abbr} ${side==='user'?su:sc} — ${opp==='user'?su:sc} ${teamInfo(opp).abbr}`,
        `TEAM PTS ${tot.pts} · THREES ${tot.fg3}`);
    });
    // 3. after 7s: back to live (game resumes from its own pause)
    this.after(7.0, ()=>{ this.endTimeout(); });
    log('TIMEOUT sequence:', side);
  },
  endTimeout(){
    gfx.telestrator(false);
    gfx.hideLowerThird();
    this.timeout = null;
    this.mode = 'live';
    this.release();
  },

  /* ---------------- Halftime show ---------------- */
  startHalftime(){
    if (this.mode !== 'live') return;
    this.mode = 'halftime';
    const su = G('scoreUser',0), sc = G('scoreCPU',0);
    const hu = teamInfo('user'), ha = teamInfo('cpu');
    this.cut(Shots.overhead(), 'halftime');
    gfx.card('NBA FINALS', 'HALFTIME', `${hu.abbr} ${su}  ·  ${sc} ${ha.abbr}`, 3400);
    sting('halftime');
    this.after(3.6, ()=>{
      const tu = stats.teamTotals('user'), tc = stats.teamTotals('cpu');
      gfx.blast('FIRST HALF RECAP', `${hu.abbr} ${su} — ${sc} ${ha.abbr}`,
        `${hu.abbr}: ${tu.pts} PTS (${tu.fg3} THREES)   ·   ${ha.abbr}: ${tc.pts} PTS (${tc.fg3} THREES)`, 4200);
    });
    this.after(8.2, ()=>{
      // leading scorer lower third
      let best=null, bestKey='';
      for (const [k,s] of stats.map){ if(!best||s.pts>best.pts){best=s;bestKey=k;} }
      if (best){
        const [tm,ix] = bestKey.split(':');
        const rn = rosterName(tm, +ix);
        gfx.lowerThird(rn.name+' · FIRST HALF', rn.number, stats.line(tm,+ix));
      }
      this.cut(Shots.celebration(), 'halftime');
    });
    this.after(12.5, ()=>{
      gfx.wipe('NBA FINALS', 'SECOND HALF', 'BACK TO LIVE ACTION', 2000);
      this.mode = 'live';
      this.release();
    });
    log('HALFTIME show started');
  },

  /* ---------------- Buzzer beater / game winner ---------------- */
  buzzerBeater(play){
    // play: {team, idx, points, loc}
    this.mode = 'buzzer';
    const rn = rosterName(play.team, play.idx);
    const ti = teamInfo(play.team);
    // freeze on the shooter, tight iso
    this.cut(Shots.isoStar(play.team, play.idx), 'buzzer');
    gfx.card('NBA FINALS', 'BUZZER BEATER', `${rn.name} · ${ti.name}`, 3000);
    sting('buzzer');
    this.after(3.2, ()=>{
      // replay it from the baseline
      this.startReplay({ label:'BUZZER BEATER', team:play.team, idx:play.idx,
        points:play.points, kind:'JUMPER', loc:play.loc, dur:5.0 });
      // chain: after replay, celebration wide, then back to live
      // (onGameEnd will take over if the game is actually over)
      const prevEnd = this.endReplay.bind(this);
      this.endReplay = ()=>{
        prevEnd();
        this.endReplay = director.endReplay; // restore
        if (this.mode !== 'buzzer') return;
        this.cut(Shots.celebration(), 'buzzer');
        this.after(4.5, ()=>{
          if (this.mode === 'buzzer'){ this.mode = 'live'; this.release(); }
        });
      };
    });
    log('BUZZER BEATER sequence');
  },
};

/* 6. Tiny synth stings (optional, off by default)                      */

let AC = null;
function sting(kind){
  if (!config.sound) return;
  try {
    AC = AC || new (window.AudioContext||window.webkitAudioContext)();
    const t0 = AC.currentTime;
    const seqs = {
      replay:   [[392,0,.16],[523,.14,.16],[659,.28,.3]],
      timeout:  [[330,0,.2],[262,.18,.35]],
      halftime: [[523,0,.18],[392,.16,.18],[330,.32,.4]],
      buzzer:   [[880,0,.5],[880,.55,.9]],
      clutch:   [[440,0,.12],[440,.14,.12]],
      break:    [[262,0,.15],[392,.13,.25]],
    };
    (seqs[kind]||seqs.break).forEach(([f,dt,dur])=>{
      const o = AC.createOscillator(), g = AC.createGain();
      o.type='triangle'; o.frequency.value=f;
      g.gain.setValueAtTime(0.0001, t0+dt);
      g.gain.exponentialRampToValueAtTime(0.22, t0+dt+0.02);
      g.gain.exponentialRampToValueAtTime(0.0001, t0+dt+dur);
      o.connect(g).connect(AC.destination);
      o.start(t0+dt); o.stop(t0+dt+dur+0.05);
    });
  } catch(e){}
}

function isClutch(){
  try {
    const gs = G('gameSeconds', 9999);
    const q = G('gameQuarter', 1);
    if (q < 4) return false;
    if (gs > config.clutchWindow) return false;
    return Math.abs(G('scoreUser',0) - G('scoreCPU',0)) <= config.clutchMargin;
  } catch(e){ return false; }
}
function maybeClutchTicker(){
  const c = isClutch();
  gfx.clutch(c);
  return c;
}

const api = {
  /* lifecycle */
  init(opts){
    Object.assign(config, opts||{});
    gfx.init();
    recorder.maxAge = config.replayLength;
    log('init', config);
    return api;
  },
  config,

  /* per-frame — call BEFORE updateCamera(dt) */
  update(dt){
    try {
      recorder.record(dt);
      director.update(dt);
      maybeClutchTicker();
      // live-game camera punctuation
      if (director.mode === 'live' && !director.active) this._liveBeats(dt);
    } catch(e){ log('update err', e.message); }
  },
  _liveT: 0,
  _liveBeats(dt){
    // subtle: after made baskets the game resets; keep default cam.
    // Reserved for future live punctuation (e.g., and-one iso).
    this._liveT += dt;
  },

  /* camera override for updateCamera() */
  hasOverride(){ return director.hasOverride(); },
  applyCamera(dt){ director.applyCamera(dt); },
  replayActive(){ return director.mode === 'replay' && !!director.replay; },

  /* ---- events ---- */
  onShotResult(shot){
    // shot: {kind, made, points, team, handler}
    if (!shot) return;
    log('shot:', shot.kind, shot.made?'MADE':'MISS', shot.points, shot.team);
    if (!shot.made){
      // missed dunk = still cut-worthy if it was posterized attempt? keep live.
      return;
    }
    stats.track(shot.team, shot.handler, 'pts', shot.points);
    if (shot.points === 3) stats.track(shot.team, shot.handler, 'fg3', 1);

    const gs = G('gameSeconds', 9999);
    const clutch = isClutch();

    // buzzer beater?
    if (gs <= 1.6 && !G('practice', false)){
      const loc = getActorPos(shot.team, shot.handler);
      director.buzzerBeater({ team:shot.team, idx:shot.handler, points:shot.points, loc });
      return;
    }

    // spectacular? -> replay
    const spectacular =
      shot.kind === 'DUNK' ||
      (shot.points === 3 && clutch) ||
      (shot.kind === 'LAYUP' && clutch && Math.random() < 0.6);

    if (spectacular && config.autoReplay && director.mode === 'live'){
      const loc = getActorPos(shot.team, shot.handler);
      const rn = rosterName(shot.team, shot.handler);
      const label = shot.kind==='DUNK' ? 'POSTER DUNK' : (shot.points===3 ? 'CLUTCH THREE' : 'AND-ONE FINISH');
      // let the make breathe for a beat, then replay
      director.after(1.1, ()=>{
        if (director.mode !== 'live') return;
        gfx.lowerThird(rn.name, rn.number, `${label} · ${stats.line(shot.team, shot.handler)}`);
        director.startReplay({ label, team:shot.team, idx:shot.handler,
          points:shot.points, kind:shot.kind, loc, dur: 4.0 });
      });
      if (clutch) sting('clutch');
      return;
    }

    // ordinary make in clutch time: quick iso + lower third, no replay
    if (clutch && director.mode === 'live'){
      const rn = rosterName(shot.team, shot.handler);
      gfx.lowerThird(rn.name, rn.number, stats.line(shot.team, shot.handler));
      director.cut(Shots.isoStar(shot.team, shot.handler), 'live');
      director.after(2.6, ()=>director.release());
    }
  },

  onTimeout(side){
    director.startTimeout(side);
  },

  onQuarter(q){
    log('quarter:', q);
    gfx.showBug('GAME&nbsp;7');
    if (q === 3) this.onHalftime();
    if (q >= 4) gfx.blast('NBA FINALS', 'FOURTH QUARTER', 'WINNING TIME', 2600);
  },

  onHalftime(){
    director.startHalftime();
  },

  onGameEnd(result){
    // result: {winner:'user'|'cpu'}
    const w = result && result.winner === 'cpu' ? 'cpu' : 'user';
    const ti = teamInfo(w);
    const su = G('scoreUser',0), sc = G('scoreCPU',0);
    director.mode = 'live';
    director.cut(Shots.celebration(), 'live');
    gfx.clutch(false);
    gfx.card('NBA FINALS', 'FINAL', `${teamInfo('user').abbr} ${su}  ·  ${sc} ${teamInfo('cpu').abbr}`, 4200);
    sting('buzzer');
    director.after(4.4, ()=>{
      gfx.blast('CHAMPIONS', ti.name, 'NBA FINALS · COURTSIDE 27', 6000);
    });
    log('game end, winner:', w);
  },

  onPossessionChange(team){
    // whip-pan energy on turnovers: quick baseline cut if it was a steal
    if (director.mode !== 'live' || director.active) return;
    // light touch: brief tighter follow handled by game cam; we just note it
    log('possession ->', team);
  },

  onFoul(side){
    if (director.mode !== 'live') return;
    // cut to FT angle; game sets freeThrowMode which its own cam handles —
    // we add the broadcast layer:
    const ti = teamInfo(side);
    gfx.blast('FOUL', ti.name, 'SHOOTING FOUL · AT THE LINE', 2200);
  },

  onBreakStart(label){
    // commercial-break style transition (e.g., long pause, quarter breaks)
    director.mode = 'break';
    gfx.wipe('NBA FINALS', label || 'TIMEOUT', 'WE\u2019LL BE RIGHT BACK', 2200);
    sting('break');
    director.after(2.4, ()=>{ director.mode='live'; });
  },
  onBreakEnd(){
    gfx.wipe('NBA FINALS', 'BACK TO THE ACTION', '', 1400);
    director.mode = 'live';
    director.release();
  },

  /* manual camera calls (coach can trigger) */
  cutTo(name, a, b){
    const s = Shots[name];
    if (!s) return;
    director.cut(s(a, b), 'live');
    director.after(4.0, ()=>director.release());
  },
  releaseCamera(){ director.release(); },

  /* stats */
  trackStat(team, idx, k, delta){ stats.track(team, idx, k, delta||1); },
  resetStats(){ stats.reset(); },

  /* jumbotron hook — game wires its canvas texture here */
  _jumboFn: null,
  onJumbotron(fn){ this._jumboFn = fn; },
  jumbotron(text){ try{ if(this._jumboFn) this._jumboFn(text); }catch(e){} },

  /* expose for debugging */
  _director: director,
  _shots: Shots,
  _gfx: gfx,
};

return api;
})();

function ensureTicker() {
  if (tickerEl) return;
  const st = document.createElement('style');
  st.textContent = TICKER_CSS;
  document.head.appendChild(st);
  tickerEl = document.createElement('div');
  tickerEl.id = 'commentaryTicker';
  lineEl = document.createElement('div');
  lineEl.className = 'cm-line';
  speakerEl = document.createElement('span');
  speakerEl.className = 'cm-speaker';
  const textSpan = document.createElement('span');
  textSpan.className = 'cm-text';
  textSpan.id = 'cmText';
  lineEl.appendChild(speakerEl);
  lineEl.appendChild(textSpan);
  tickerEl.appendChild(lineEl);
  document.body.appendChild(tickerEl);
  textEl = textSpan;
}

/* ---------------- Queue (never overlaps) ---------------- */
const queue = [];          // pending lines
let speaking = false;      // a line is on screen
let lastLineKey = '';      // anti-repeat
const recentKeys = [];     // short memory to avoid repeats

function keyOf(speaker, text) { return speaker + '|' + text; }

function remember(k) {
  recentKeys.push(k);
  if (recentKeys.length > 14) recentKeys.shift();
}

/* Weighted pick with anti-repeat */
function pick(arr) {
  const fresh = arr.filter(t => !recentKeys.includes(t));
  const pool = fresh.length ? fresh : arr;
  return pool[Math.floor(Math.random() * pool.length)];
}

function fmt(s, vars) {
  return s.replace(/\{(\w+)\}/g, (_, k) => (vars[k] !== undefined ? vars[k] : ''));
}

/* Say a line. opts: {speaker:'pbp'|'color', big:bool, priority:bool} */
function say(text, speaker, opts) {
  opts = opts || {};
  const line = { text, speaker: speaker || 'pbp', big: !!opts.big };
  if (opts.priority) {
    // Big moments cut to the front of the line (but never mid-sentence)
    queue.unshift(line);
  } else {
    queue.push(line);
  }
  pump();
}

function lineDurationMs(text) {
  // ~ speaking pace, clamped
  return Math.min(6500, Math.max(2200, 900 + text.length * 42));
}

function pump() {
  if (speaking || !queue.length) return;
  ensureTicker();
  speaking = true;
  const line = queue.shift();
  const booth = BOOTH[line.speaker] || BOOTH.pbp;
  remember(keyOf(line.speaker, line.text));
  lastLineKey = keyOf(line.speaker, line.text);

  speakerEl.textContent = booth.name + ':';
  speakerEl.className = 'cm-speaker ' + booth.cls;
  // Support BANG! marker for emphasized words
  textEl.innerHTML =
    '<span class="cm-fade">' + line.text.replace(/BANG!/g, '<span class="bang">BANG!</span>') + '</span>';
  lineEl.classList.toggle('big', line.big);
  tickerEl.classList.add('on');

  setTimeout(() => {
    tickerEl.classList.remove('on');
    setTimeout(() => { speaking = false; pump(); }, 380);
  }, lineDurationMs(line.text));
}

/* Convenience: pbp says, color answers (classic booth exchange) */
function exchange(pbpText, colorText, big) {
  say(pbpText, 'pbp', { big: !!big });
  if (colorText) say(colorText, 'color');
}

/* ---------------- Game state tracking (for color commentary) ---------------- */
const teams = {}; // abbr -> {name, fgM,fgA, t3M,t3A, run, oppRun, lastShots:[bool], pts}

function team(abbr, name) {
  if (!teams[abbr]) teams[abbr] = { abbr, name: name || abbr, fgM:0, fgA:0, t3M:0, t3A:0, run:0, oppRun:0, last:[] };
  if (name) teams[abbr].name = name;
  return teams[abbr];
}

function recordShot(abbr, name, made, isThree) {
  const t = team(abbr, name);
  t.fgA++; if (made) t.fgM++;
  if (isThree) { t.t3A++; if (made) t.t3M++; }
  t.last.push(made); if (t.last.length > 12) t.last.shift();
  return t;
}

function recordScore(abbr, pts) {
  for (const k in teams) {
    if (k === abbr) { teams[k].run += pts; teams[k].oppRun = 0; }
    else { teams[k].oppRun += pts; teams[k].run = 0; }
  }
}

function threePct(t) { return t.t3A ? Math.round(100 * t.t3M / t.t3A) : 0; }
function lastN(t, n) { return t.last.slice(-n); }
function coldFromThree(t) { const l = lastN(t, 6).filter((_, i, a) => true); return t.t3A >= 5 && threePct(t) < 25; }
